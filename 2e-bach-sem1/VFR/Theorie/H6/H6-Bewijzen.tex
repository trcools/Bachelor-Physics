\section{Multilinear Algebra}

\begin{res}{Proposition}{6.1}
If different pairs $(W,\chi)$ and $(W',\chi')$ satisfy this definition, then there exists an isomorphism $\phi\in\operatorname{Hom}(W,W')$ such that $\chi'=\phi\circ\chi$, and thus $W\cong W'$.
\end{res}

\begin{res}{Proposition}{6.2}
Consider three vector spaces $U$, $V$, $W$ over the same field $\mathbb F$. The nested tensor product spaces $(U\otimes V)\otimes W$ and $U\otimes(V\otimes W)$ are isomorphic.
\end{res}

\begin{res}{Proposition}{6.3}
For an inner product space $V$ and vectors $v_i,w_i\in V$ for $i=1,\dots,p$, we find
\begin{multline}\langle w_1\wedge w_2\wedge\dots\wedge w_p,v_1\wedge v_2\wedge\dots\wedge v_p\rangle=\\
\det\left(\begin{bmatrix}\langle w_1,v_1\rangle&\langle w_1,v_2\rangle&\ldots&\langle w_1,v_p\rangle\\\langle w_p,v_1\rangle&\langle w_2,v_2\rangle&\ldots&\langle w_2,v_p\rangle\\\vdots&\vdots&\ddots&\vdots\\\langle w_p,v_1\rangle&\langle w_p,v_2\rangle&\ldots&\langle w_p,v_p\rangle\end{bmatrix}\right)\tag{6.31}\end{multline}
\end{res}
