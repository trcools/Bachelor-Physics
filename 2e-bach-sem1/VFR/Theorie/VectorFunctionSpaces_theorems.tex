\section{Elementary Algebraic Structures}

\begin{res}{Proposition}{1.1}
If $\ker\varphi=\{e_G\}$, then $\varphi$ is injective.
\end{res}

\begin{res}{Proposition}{1.2}
The kernel of a group homomorphism $\varphi:G\to H$ is a subgroup of $G$.
\end{res}

\begin{res}[Quotient group]{Proposition}{1.3}
For a group $G$ and normal subgroup $N$, the relation $g\sim h$ if $\exists n\in N$ such that $g\cdot n=h$ is an equivalence relation. The quotient set $G/\!\sim$ is typically denoted as $G/N$ and can be given the structure of a group, referred to as the quotient group.
\end{res}

\begin{res}{Proposition}{1.4}
In a finite ring, all elements must either have multiplicative inverses, or must be zero divisors.
\end{res}

\begin{res}{Proposition}{1.5}
For a set $S\subseteq V$ that is linearly independent, the decomposition of any vector $v\in\operatorname{span}(S)$ as a linear combination of the form Eq.~(1.9) is unique.
\end{res}

\begin{res}{Proposition}{1.6}
Let $S\subseteq V$ be linearly independent, such that no $v\in V$ can be added to $S$ without making it linearly dependent. Then $\operatorname{span}(S)=V$ and $S$ is a basis of $V$.
\end{res}

\begin{res}{Proposition}{1.7}
From any complete set $S\subseteq V$, we can extract a subset that is a basis for $V$.
\end{res}

\begin{res}{Proposition}{1.8}
Every vector space admits a basis.
\end{res}

\begin{res}{Proposition}{1.9}
Any linearly independent set $S\subseteq V$ can be extended to be a basis for $V$, i.e.\ there exists a set $S'\subseteq V$ such that $V=\mathbb F(S\cup S')$.
\end{res}

\begin{res}{Proposition}{1.10}
Let $S=\{v_1,\dots,v_m\}$ be a linearly independent subset of $V$, then $m\le\dim(V)=n$, i.e.\ an $n$-dimensional space does not admit more than $n$ linearly independent vectors.
\end{res}

\begin{res}{Corollary}{1.11}
Any basis $B$ of a finite-dimensional vector space $V$ has $|B|=\dim(V)$, and any linearly independent set $S$ with $|S|=\dim(V)$ is complete and thus a basis.
\end{res}

\begin{res}{Proposition}{1.12}
Any two finite-dimensional vector spaces $V$ and $W$ with $\dim(V)=\dim(W)=n$ are isomorphic, henceforth denoted as $V\cong W$.
\end{res}

\begin{res}{Proposition}{1.13}
The relation $W\preccurlyeq V$ implies that $\dim(W)\le\dim(V)$.
\end{res}

\begin{res}{Proposition}{1.14}
Let $W\preccurlyeq V$ where $V$ (and thus also $W$) are finite-dimensional; then $\dim(W)=\dim(V)$ implies $W=V$
\end{res}

\begin{res}{Proposition}{1.15}
Given two subspaces $W_i\preccurlyeq V$, $i=1,2$, then their \textbf{intersection} (by just interpreting them as sets) is still a subspace $W_1\cap W_2\preccurlyeq V$.
\end{res}

\begin{res}{Proposition}{1.16}
If $W_1$ and $W_2$ are finite-dimensional, then
\begin{equation}\dim(W_1+W_2)=\dim(W_1)+\dim(W_2)-\dim(W_1\cap W_2).\tag{1.16}\end{equation}
\end{res}

\begin{res}{Proposition}{1.17}
Given a subspace $W\preccurlyeq V$, there always exists a complementary space $U\preccurlyeq V$ such that $V=U\oplus W$.
\end{res}

\begin{res}{Proposition}{1.18}
For two vector spaces $W_1$ and $W_2$ over a common field $\mathbb F$ the Cartesian product $W_1\times W_2=\{(w_1,w_2),\forall w_1\in W_1,w_2\in W_2\}$ can be made into a vector space over $\mathbb F$ by defining vector addition as $(v_1,v_2)+(w_1,w_2)=(w_1+v_1,w_2+v_2)$ and scalar multiplication as $a(w_1,w_2)=(aw_1,aw_2)$, $\forall a\in\mathbb F$.
\end{res}

\begin{res}{Proposition}{1.19}
If $W_1$ and $W_2$ are two complementary subspaces of some bigger space $V=W_1\oplus W_2$, then the (internal) direct sum construction is isomorphic to $W_1\times W_2$.
\end{res}

\begin{res}{Proposition}{1.20}
Given a subspace $W\preccurlyeq V$, the quotient set $V/\!\sim$ under the equivalence relation $v_1\sim v_2$ if $v_1-v_2\in W$ can be given the structure of a vector space, referred to as the \textbf{quotient space} $V/W$, by defining addition and scalar multiplication of the equivalence classes as
\begin{equation}(v_1+W)+(v_2+W)=(v_1+v_2+W),\qquad a(v+W)=(av+W).\tag{1.17}\end{equation}
\end{res}

\begin{res}{Proposition}{1.21}
$\dim(V/W)=\dim(V)-\dim(W)$
\end{res}

\begin{res}{Corollary}{1.22}
A direct consequence of this proof is also that all complements of a given subspace $W\preccurlyeq V$ are isomorphic to $W/V$, and thus also to each other.
\end{res}

\section{Linear Maps and Matrices}

\begin{res}{Proposition}{2.1}
The set of linear maps $\operatorname{Hom}(V,W)$ can itself be given the structure of a vector space by introducing an addition and scalar multiplication operators in terms of those from the codomain $W$ as follows:
\begin{itemize}
\item $\forall\hat A,\hat B\in\operatorname{Hom}(V,W)$, $(\hat A+\hat B):V\to W:v\mapsto(\hat A+\hat B)(v)=\hat A(v)+\hat B(v)$;
\item $\forall\hat A\in\operatorname{Hom}(V,W)$, $\forall a\in\mathbb F$, $(a\hat A):V\to W:v\mapsto(a\hat A)(v)=a(\hat A(v))$.
\end{itemize}
The additive neutral element is the \textbf{zero map} $\hat 0:V\to W$, which maps any $v\in V$ to $o\in W$.
\end{res}

\begin{res}{Proposition}{2.2}
The composition operator
\[\circ:\operatorname{Hom}(V,W)\times\operatorname{Hom}(U,V)\to\operatorname{Hom}(U,W):(\hat A,\hat B)\to\hat A\circ\hat B\]
is bilinear with respect to this addition and scalar multiplication on the Hom spaces.
\end{res}

\begin{res}{Corollary}{2.3}
The space of linear operators $\operatorname{End}(V)$ in combination with the composition operator $\circ$ has the structure of an (associative) algebra (and thus of a ring) with the \textbf{identity map} $\hat 1=\hat 1_V=\mathrm{id}_V:v\mapsto v$ as multiplicative unit.
\end{res}

\begin{res}{Proposition}{2.4}
The kernel of $\hat A\in\operatorname{Hom}(V,W)$ has the structure of a subspace of $V$, i.e.\ $\ker(\hat A)=\hat A^{-1}(o_W)\preccurlyeq V$. It is referred to as the \textbf{null space} (\textit{nulruimte}) of $\hat A$.
\end{res}

\begin{res}{Proposition}{2.5}
The \textbf{image} (\textit{beeld} of \textit{beeldruimte}) of the linear map $\hat A:V\to W$ also has the structure of a subspace of $W$, i.e.\ $\operatorname{im}(\hat A)=\hat A(V)\preccurlyeq W$.
\end{res}

\begin{res}{Corollary}{2.6}
More generally, a linear map $\hat A:V\to W$ maps subspaces $U\preccurlyeq V$ to subspaces $\hat AU\preccurlyeq W$.
\end{res}

\begin{res}{Proposition}{2.7}
A map $\hat A\in\operatorname{Hom}(V,W)$ is injective if and only if $\operatorname{nullity}(\hat A)=0$.
\end{res}

\begin{res}{Proposition}{2.8}
Injective linear maps $\hat A$ preserve linear independence of vectors, and thus, dimensionality of subspaces, i.e.\ for any subspace $U\preccurlyeq V$, $\dim(U)=\dim(\hat AU)$.
\end{res}

\begin{res}{Proposition}{2.9}
For finite-dimensional $W$, a map $\hat A\in\operatorname{Hom}(V,W)$ is surjective if and only if $\operatorname{rank}(\hat A)=\dim(W)$.
\end{res}

\begin{res}{Proposition}{2.10}
For $\hat A\in\operatorname{Hom}(V,W)$, $\operatorname{im}(\hat A)$ is isomorphic to $V/\ker(\hat A)$.
\end{res}

\begin{res}[Rank-nullity theorem]{Theorem}{2.11}
For $\hat A\in\operatorname{Hom}(V,W)$ it holds that
\[\operatorname{rank}(\hat A)+\operatorname{nullity}(\hat A)=\dim(\operatorname{im}(\hat A)+\dim(\ker(\hat A))=\dim(V)=\dim(\operatorname{dom}(\hat A)).\]
\end{res}

\begin{res}{Corollary}{2.12}
For $\hat A\in\operatorname{Hom}(V,W)$ with $V$ and $W$ finite-dimensional and $\dim(V)=\dim(W)$, the following properties are equivalent
\begin{itemize}
\item $\hat A$ is injective.
\item $\hat A$ is surjective.
\item $\hat A$ is bijective.
\end{itemize}
\end{res}

\begin{res}{Proposition}{2.13}
The row and column rank of a matrix coincide, or thus, $\operatorname{rank}(\mathsf A)=\operatorname{rank}(\mathsf A^{\mathsf T})$.
\end{res}

\begin{res}{Proposition}{2.14}
For two matrices $\mathsf A$ and $\mathsf B$, we have $\operatorname{rank}(\mathsf{AB})\le\operatorname{rank}(\mathsf A)$ and $\operatorname{rank}(\mathsf{AB})\le\operatorname{rank}(\mathsf B)$.
\end{res}

\begin{res}{Proposition}{2.15}
The (straightforward) evaluation the multiplication of a matrix $\mathsf A\in\mathbb F^{m\times p}$ with a matrix $\mathsf B\in\mathbb F^{p\times n}$ in order to produce a matrix $\mathsf C=\mathsf{AB}\in\mathbb F^{m\times n}$ requires $mnp$ scalar multiplications (and $mn(p-1)$ scalar additions).
\end{res}

\begin{res}{Proposition}{2.16}
For $\mathsf A,\mathsf B\in\mathbb F^{n\times n}$, the trace has the cyclic property $\operatorname{tr}(\mathsf{AB})=\operatorname{tr}(\mathsf{BA})$
\end{res}

\begin{res}{Proposition}{2.17}
When interpreting the determinant as a function of the $n$ columns $\boldsymbol a_j=(A^i{}_j)_{i=1,\dots,n}$ of $\mathsf A$, it is uniquely characterized by the following three properties
\begin{itemize}
\item The determinant is linear in each of the $n$ columns, i.e.\ it is a multilinear function of $(\boldsymbol a_1,\boldsymbol a_2,\dots,\boldsymbol a_n)$.
\item The determinant is alternating in any two subsequent arguments, i.e.\ if $\boldsymbol a_j=\boldsymbol a_{j+1}$ for any $j=1,\dots,n-1$, then $\det(\mathsf A)=0$.
\item The determinant of the identity matrix, or thus of the standard basis $\boldsymbol a_j=\boldsymbol e_j$, is $\det(\mathsf I_n)=1$.
\end{itemize}
\end{res}

\begin{res}{Proposition}{2.18}
For a square matrix $\mathsf A\in\mathbb F^{n\times n}$, $\det(\mathsf A)=0$ if and only if $\operatorname{rank}(\mathsf A)<n$.
\end{res}

\begin{res}{Lemma}{2.19}
\begin{equation}\epsilon^{j_1j_2\ldots j_n}A^{i_1}{}_{j_1}A^{i_2}{}_{j_2}\ldots A^{i_n}{}_{j_n}=\det(\mathsf A)\,\epsilon^{i_1i_2\ldots i_n}.\tag{2.12}\end{equation}
\end{res}

\begin{res}{Theorem}{2.20}
For two matrices $\mathsf A,\mathsf B\in\mathbb F^{n\times n}$, we have $\det(\mathsf{AB})=\det(\mathsf A)\det(\mathsf B)$.
\end{res}

\begin{res}{Proposition}{2.21}
For $\mathsf A\in\mathbb F^{n\times n}$, $\det(\mathsf A^{\mathsf T})=\det(\mathsf A)$ and $\det(\mathsf A^{\mathsf H})=\overline{\det(\mathsf A)}$.
\end{res}

\begin{res}{Proposition}{2.22}
An explicit expression for the $(k,l)$-minor $M_k{}^l$ of the matrix $\mathsf A\in\mathbb F^{n\times n}$ is given by
\begin{equation}M_k{}^l=\sum_{j_1,\dots,j_{k-1},j_{k+1},\dots,j_n}(-1)^{l-k}\epsilon^{j_1j_2\ldots j_{k-1}\,l\,j_{k+1}\ldots j_n}A^1{}_{j_1}\ldots A^{k-1}{}_{j_{k-1}}A^{k+1}{}_{j_{k+1}}\ldots A^n{}_{j_n}\tag{2.18}\end{equation}
\end{res}

\begin{res}[Laplace expansion]{Corollary}{2.23}
It now follows trivially that the determinant of $\mathsf A$ can be written as
\begin{equation}\det(\mathsf A)=\sum_l A^k{}_l(-1)^{k-l}M_k{}^l\quad\text{(no summation over $k$).}\tag{2.19}\end{equation}
\end{res}

\begin{res}{Proposition}{2.24}
Let $\mathsf A\in\mathbb F^{n\times n}$. We have
\begin{equation}\mathsf A\cdot\operatorname{adj}(\mathsf A)=\det(\mathsf A)\mathsf I_n\tag{2.22}\end{equation}
or thus, whenever $\det(\mathsf A)\ne0$.
\begin{equation}\mathsf A^{-1}=\det(\mathsf A)^{-1}\operatorname{adj}(\mathsf A).\tag{2.23}\end{equation}
\end{res}

\begin{res}{Proposition}{2.25}
The matrix inverse has the following properties
\[\det(\mathsf A^{-1})=\det(\mathsf A)^{-1},\qquad(\mathsf A^{-1})^{-1}=\mathsf A,\qquad(\mathsf{AB})^{-1}=\mathsf B^{-1}\mathsf A^{-1}.\]
\end{res}

\begin{res}[Jacobi's formula]{Proposition}{2.26}
Given a one-parameter family of square matrices $\mathsf A(t)\in\mathbb F^{n\times n}$, it holds that
\begin{equation}\frac{\mathrm d}{\mathrm dt}\det\bigl(\mathsf A(t)\bigr)=\det\bigl(\mathsf A(t)\bigr)\operatorname{tr}\!\left(\mathsf A(t)^{-1}\frac{\mathrm d\mathsf A}{\mathrm dt}(t)\right).\tag{2.24}\end{equation}
\end{res}

\begin{res}{Proposition}{2.27}
On the space $\mathbb F^{n\times n}$ of square matrices, being related by a similarity transform, i.e.\ $\mathsf A\sim\mathsf B$ if $\exists\mathsf T\in\mathrm{GL}(n,\mathbb F)$ such that $\mathsf B=\mathsf{TAT}^{-1}$ is an equivalence relation.
\end{res}

\begin{res}{Proposition}{2.28}
The matrix trace and determinant are basis independent.
\end{res}

\begin{res}{Proposition}{2.29}
For a finite-dimensional vector space $V$, a choice of a basis $B=\{e_1,\dots,e_n\}$ induces a canonical basis for $V^*$, which we denote as $B^*=\{\varepsilon^1,\dots,\varepsilon^n\}$ and refer to as the \textbf{dual basis}. It is defined via
\begin{equation}\varepsilon^i[e_j]=\delta^i{}_j.\tag{2.30}\end{equation}
\end{res}

\begin{res}{Proposition}{2.30}
Consider $\xi,\chi\in V^*$. We have $\ker(\xi)=\ker(\chi)$ if and only if there exists a non-zero scalar $a\in\mathbb F$ for which $\xi=a\chi$.
\end{res}

\begin{res}{Proposition}{2.31}
The inverse of an upper/lower triangular matrix $\mathsf A$ is upper/lower triangular.
\end{res}

\section{Linear Operators and Eigenvalues}

\begin{res}{Proposition}{3.1}
Let $p,q\in\mathbb F[x]$ be two arbitrary univariate polynomials and $\hat A\in\operatorname{End}V$ be a linear operator on $V$. We have
\begin{equation}\bigl[p(\hat A),q(\hat A)\bigr]=\hat 0\tag{3.4}\end{equation}
\end{res}

\begin{res}{Proposition}{3.2}
Let $p\in\mathbb F[x]$ be an arbitrary univariate polynomial, $\hat A\in\operatorname{End}V$ a linear operator on $V$ and $\hat G\in\mathrm{GL}(V)\subseteq\operatorname{End}(V)$ and invertible linear operator. We have
\begin{equation}p(\hat G^{-1}\hat A\hat G)=\hat G^{-1}p(\hat A)\hat G\tag{3.5}\end{equation}
\end{res}

\begin{res}{Proposition}{3.3}
Let $\hat P$ be an idempotent operator on $V$, then $V=\operatorname{im}(\hat P)\oplus\ker(\hat P)$.
\end{res}

\begin{res}{Proposition}{3.4}
For a given linear operator $\hat A\in\operatorname{End}(V)$ where $V$ is a finite-dimensional vector space, there exists a unique monic polynomial of lowest degree $s$ that annihilates $\hat A$. This polynomial is known as the \textbf{minimal annihilating polynomial} $m_{\hat A}(x)=x^s+\dots$.
\end{res}

\begin{res}{Proposition}{3.5}
For a given operator $\hat A\in\operatorname{End}(V)$, if $p\in\mathbb F[x]$ annihilates $\hat A$, then $p(x)=m_{\hat A}(x)q(x)$ for some other polynomial $q\in\mathbb F[x]$.
\end{res}

\begin{res}{Proposition}{3.6}
Consider a linear operator $\hat A\in\operatorname{End}(V)$, an eigenvalue-eigenvector pair $(\lambda,v)$ and a polynomial $p\in\mathbb F[x]$. It follows that $p(\hat A)v=p(\lambda)v$.
\end{res}

\begin{res}{Corollary}{3.7}
If $p\in\mathbb F[x]$ is an annihilating polynomial of $\hat A$, any eigenvalue $\lambda$ of $\hat A$ is a root of $p$, i.e.\ $p(\lambda)=0$. In particular, any eigenvalue $\lambda$ is a root of the minimal annihilating polynomial $m_{\hat A}(z)$.
\end{res}

\begin{res}{Proposition}{3.8}
A set of eigenvectors $\{v_i;\ i=1,\dots,k\}$ of a linear operator $\hat A$ associated with mutually distinct eigenvalues $\lambda_i$, for $i=1,\dots,k$ (that is, $\lambda_i\ne\lambda_j$ for $i\ne j$) is linearly independent.
\end{res}

\begin{res}{Proposition}{3.9}
For a set of eigenspaces $\{V_{\lambda_i};\ i=1,\dots,k\}$ of a linear operator $\hat A$ associated with mutually distinct eigenvalues $\lambda_i$, for $i=1,\dots,k$ (that is, $\lambda_i\ne\lambda_j$ for $i\ne j$), the sum $\sum_{i=1}^kV_{\lambda_i}$ is actually a direct sum $\bigoplus_{i=1}^kV_{\lambda_i}$.
\end{res}

\begin{res}{Corollary}{3.10}
Let $V$ be a finite-dimensional vector space and $\hat A$ a linear operator on $V$ with eigenvalues $\sigma_{\hat A}=\{\lambda_1,\dots,\lambda_m\}$ and corresponding eigenspaces $V_{\lambda_i}$ for $i=1,\dots,m$. Because $\bigoplus_{\lambda\in\sigma_{\hat A}}V_\lambda\preccurlyeq V$, we find
\begin{equation}\sum_{\lambda\in\sigma_{\hat A}}r_\lambda\le\dim(V).\tag{3.12}\end{equation}
\end{res}

\begin{res}{Proposition}{3.11}
Consider linear maps $\hat A\in\operatorname{Hom}(V,W)$ and $\hat B\in\operatorname{Hom}(W,V)$ for two vector space $V$ and $W$ (which could be equal). Any \textbf{nonzero} eigenvalue $\lambda$ of $\hat B\hat A$ with eigenspace $V_\lambda$ is also an eigenvalue of $\hat A\hat B$ with eigenspace $W_\lambda=\hat AV_\lambda$, and with $\dim(V_\lambda)=\dim(W_\lambda)$.
\end{res}

\begin{res}{Proposition}{3.12}
Let $p(z)=z^n+p_{n-1}z^{n-1}+\ldots+p_0$ be a monic polynomial of degree $n$. The Frobenius \textbf{companion matrix} $\mathsf C_p\in\mathbb F^{n\times n}$ associated with $p$ is given by
\begin{equation}\mathsf C_p=\begin{bmatrix}0&1&0&\ldots&0&0\\0&0&1&\ldots&0&0\\\vdots&\vdots&\vdots&\ddots&\vdots&\vdots\\0&0&0&\ldots&0&1\\-p_0&-p_1&-p_2&\ldots&-p_{n-2}&-p_{n-1}\end{bmatrix}\tag{3.17}\end{equation}
and satisfies $k_{\mathsf C_p}(z)=\det(z\mathsf I-\mathsf C_p)=p(z)$.
\end{res}

\begin{res}[Cayley-Hamilton]{Theorem}{3.13}
The characteristic polynomial $k_{\hat A}(z)$ of a linear operator $\hat A\in\operatorname{End}(V)$ annihilates $\hat A$.
\end{res}

\begin{res}{Proposition}{3.14}
Let $V$ be a finite-dimensional vector space and $\hat A$ a linear operator on $V$ with eigenvalues $\sigma_{\hat A}=\{\lambda_1,\dots,\lambda_m\}$. The operator $\hat A$ is \textbf{diagonalisable} if and only if the minimal annihilating polynomial takes the form
\begin{equation}m_{\hat A}(z)=\prod_{\lambda\in\sigma_{\hat A}}(z-\lambda),\tag{3.27}\end{equation}
i.e.\ if $s_{\lambda_i}=1$ for all $i=1,\dots,m$.
\end{res}

\begin{res}{Corollary}{3.15}
A defective operator has $s_\lambda>1$ for at least one eigenvalue $\lambda$.
\end{res}

\begin{res}{Corollary}{3.16}
For an operator $\hat A\in\operatorname{End}(V)$ with $\dim(V)=n$, the property of $k_{\hat A}(z)$ having $n$ distinct roots is a sufficient criterion for being diagonalisable, since this implies that $q_\lambda=1$ and thus also $s_\lambda$ since $s_\lambda\le q_\lambda$.
\end{res}

\begin{res}{Theorem}{3.17}
Let $\hat A,\hat B\in\operatorname{End}(V)$ be diagonalisable operators that satisfy $[\hat A,\hat B]=\hat 0$. They then admit a common spectral decomposition, i.e.\ there exists a set of projectors $\{\hat P_i;i\in I\}$ with $I$ some indexing set, such that
\begin{equation}\hat P_i\hat P_j=\delta_{i,j}\hat P_i,\qquad\sum_{i\in I}\hat P_i=\hat 1_V,\qquad\hat A=\sum_{i\in I}\lambda_i\hat P_i\qquad\hat B=\sum_{i\in I}\mu_i\hat P_i\tag{3.33}\end{equation}
\end{res}

\begin{res}{Lemma}{3.18}
If $U\preccurlyeq V$ is an invariant subspace of $\hat A\in\operatorname{End}(V)$, it is also an invariant subspace of $\hat A+a\hat 1$ for any $a\in\mathbb F$.
\end{res}

\begin{res}{Proposition}{3.19}
For any operator $\hat A$ on a finite-dimensional space $V$, any subspace in the following two sequences of nested subspaces
\begin{align*}
\{o\}&=\ker((\hat A-a\hat 1)^0)\preccurlyeq\ker(\hat A-a\hat 1)\preccurlyeq\ker((\hat A-a\hat 1)^2)\preccurlyeq\ldots\preccurlyeq\ker((\hat A-a\hat 1)^k)\preccurlyeq\ldots\\
V&=\operatorname{im}((\hat A-a\hat 1)^0)\succcurlyeq\operatorname{im}(\hat A-a\hat 1)\succcurlyeq\operatorname{im}((\hat A-a\hat 1)^2)\succcurlyeq\ldots\succcurlyeq\operatorname{im}((\hat A-a\hat 1)^k)\succcurlyeq\ldots
\end{align*}
is an invariant subspace for any scalar $a\in\mathbb F$. Furthermore, the dimension of the spaces $\ker((\hat A-a\hat 1)^k)$ is strictly increasing up to a value $k=s$, after which it remains constant for any $k\ge s$, thus implying that the first sequence stabilises, i.e.\ all subspaces for $k\ge s$ are the same. At the same time, the dimension of $\operatorname{im}((\hat A-a\hat 1)^k)$ is strictly decreasing up to $k=s$, after which the second sequence stabilises. Finally, for any $k\ge s$, we have $V=\ker((\hat A-a\hat 1)^k)\oplus\operatorname{im}((\hat A-a\hat 1)^k)$.
\end{res}

\begin{res}{Theorem}{3.20}
Let $\hat A$ be a linear operator on a finite-dimensional vector space $V$. We can construct a direct sum decomposition of $V$ as
\begin{equation}V=\bigoplus_{\lambda\in\sigma_{\hat A}}U_\lambda=U_{\lambda_1}\oplus U_{\lambda_2}\oplus\ldots\oplus U_{\lambda_m}\tag{3.45}\end{equation}
in terms of the generalised eigenspaces $U_\lambda=\ker((\hat A-\lambda\hat 1)^{s_\lambda})$ with $s_\lambda$ the exponent of the factor $(z-\lambda)$ in the minimal annihilating polynomial $m_{\hat A}(z)$. Each component $U_\lambda$ is an invariant subspace of $\hat A$ and has dimension $\dim(U_\lambda)=q_\lambda$, the algebraic multiplicity, or thus, the exponent of $(z-\lambda)$ in the characteristic polynomial $k_{\hat A}(z)$. The restriction of $\hat A$ to $U_\lambda$ takes the form
\begin{equation}\hat A|_{U_\lambda}=\lambda\hat 1_{U_\lambda}+\hat N_\lambda\tag{3.46}\end{equation}
with $\hat N_\lambda$ a nilpotent operator on $U_\lambda$ with index $s_\lambda$.
\end{res}

\begin{res}{Proposition}{3.21}
Let $\hat N$ be a nilpotent operator of index $s$ on a space $U$ with $\dim(U)=q$. There exists a basis $B$ for $U$ such that $\mathsf N=\Phi_B(\hat N)=\bigoplus_{k=1}^s\bigoplus_{i=1}^{p_k}\mathsf N_k$, where the $k\times k$ matrices $\mathsf N_k$ take the canonical form
\begin{equation}\mathsf N_k=\begin{bmatrix}0&1&0&\ldots&0&0\\0&0&1&\ldots&0&0\\\vdots&\vdots&\vdots&\ddots&\vdots&\vdots\\0&0&0&\ldots&0&1\\0&0&0&\ldots&0&0\end{bmatrix}\tag{3.52}\end{equation}
and are nilpotent with index $k$, i.e.\ they satisfy $(\mathsf N_k)^k=\mathsf O$.
\end{res}

\begin{res}[Jordan decomposition]{Theorem}{3.22}
Any linear operator $\hat A\in\operatorname{End}(V)$ on a finite-dimensional space $V$ over $\mathbb C$ admits a choice of basis $B$, known as a Jordan basis, so that it acquires a block diagonal matrix representation
\begin{equation}\mathsf J_{\hat A}=\Phi_B(\hat A)=\bigoplus_{\lambda\in\sigma_{\hat A}}\bigoplus_{k=1}^{s_\lambda}\bigoplus_{i=1}^{p_{\lambda,k}}\mathsf J_k(\lambda)\qquad\text{(with $p_{\lambda,k}\ge0$ for $k=1,\dots,s_\lambda-1$ and $p_{\lambda,s_\lambda}>0$)}\tag{3.56}\end{equation}
known as the \textbf{Jordan canonical form} (a.k.a.\ Jordan normal form), in which the blocks on the diagonal take the form of the elementary Jordan blocks from Definition 3.15. Equivalently, for any matrix $\mathsf A\in\mathbb C^{n\times n}$, there exists a basis transform $\mathsf V\in\mathrm{GL}(n,\mathbb C)$ that brings $\mathsf A$ into the Jordan canonical form
\begin{equation}\mathsf A=\mathsf V\mathsf J_{\mathsf A}\mathsf V^{-1}=\mathsf V\left[\bigoplus_{\lambda\in\sigma_{\mathsf A}}\bigoplus_{k=1}^{s_\lambda}\bigoplus_{i=1}^{p_{\lambda,k}}\mathsf J_k(\lambda)\right]\mathsf V^{-1},\tag{3.57}\end{equation}
which is also referred to as the \textbf{Jordan decomposition}.
\end{res}

\begin{res}{Proposition}{3.23}
Consider a companion matrix $\mathsf C_p\in\mathbb C^{n\times n}$ associated with an $n$th degree monic polynomial $p\in\mathbb C[z]$, as defined in Eq.~(3.17) of Proposition 3.12. The matrix $\mathsf C_p$ is non-derogatory, i.e.\ every root $\lambda$ of $p$ defines a one-dimensional eigenspace (thus $r_\lambda=1$), and a single Jordan block of order $q_\lambda$, the algebraic multiplicity defined by $p$ itself. The generalised eigenvectors take the form
\begin{equation}(\boldsymbol u_{\lambda,k})^i=\frac1{k!}\frac{\mathrm d^k}{\mathrm d\lambda^k}\lambda^{i-1}=\begin{cases}\binom{i-1}{k}\lambda^{i-1-k},&k<i\\0,&k\ge i\end{cases}\tag{3.58}\end{equation}
for $i=1,\dots,n$ and $k=0,\dots,q_\lambda-1$ with $\binom{i-1}{k}=\frac{(i-1)!}{k!(i-k-1)!}$ the binomial coefficients. In particular, $\boldsymbol v_\lambda=\boldsymbol u_{\lambda,0}=(1,\lambda,\lambda^2,\dots,\lambda^{n-1})$ corresponds to the eigenvector.
\end{res}

\begin{res}{Theorem}{3.24}
Consider a scalar-valued, linear, $k$th order recurrence relation
\begin{equation}x_{n+k}=a_0x_n+a_1x_{n+1}+\dots+a_{k-2}x_{n+k-2}+a_{k-1}x_{n+k-1}.\tag{3.85}\end{equation}
A general solution can be written as a linear combination of $k$ elementary solutions, where with every root $\lambda$ of multiplicity $q_\lambda$ of the polynomial
\begin{equation}p(z)=z^k-a_{k-1}z^{k-1}-a_{k-2}z^{k-2}-\dots-a_1z-a_0\tag{3.86}\end{equation}
gives rise to $q_\lambda$ elementary solutions of the form
\begin{equation}x_n=n^j\lambda^n,\quad j=0,\dots,q_\lambda-1.\tag{3.87}\end{equation}
The expansion coefficients of a general trajectory in terms of these elementary solutions are completely fixed by specifying $k$ initial values $(x_{-k+1},x_{-k+2},\dots,x_0)$.
\end{res}

\begin{res}{Theorem}{3.25}
Consider a scalar-valued, linear, $k$th order differential equation
\begin{equation}x^{(k)}(t)=a_0x(t)+a_1\dot x(t)+\dots+a_{k-2}x^{(k-2)}(t)+a_{k-1}x^{(k-1)}(t).\tag{3.95}\end{equation}
A general solution can be written as a linear combination of $k$ elementary solutions, where with every root $\lambda$ of multiplicity $q_\lambda$ of the polynomial
\begin{equation}p(z)=z^k-a_{k-1}z^{k-1}-a_{k-2}z^{k-2}-\dots-a_1z-a_0\tag{3.96}\end{equation}
gives rise to $q_\lambda$ elementary solutions of the form
\begin{equation}x(t)=t^j\mathrm e^{\lambda t},\quad j=0,\dots,q_\lambda-1.\tag{3.97}\end{equation}
The expansion coefficients of a general solution are completely fixed by specifying $k$ initial values $\bigl(x(0),\dot x(0),\ddot x(0),\dots,x^{(k-1)}(0)\bigr)$.
\end{res}

\section{Norms and Distances}

\begin{res}{Lemma}{4.1}
Any two nonnegative numbers $a,b\in\mathbb R_{\ge0}$ satisfy \textbf{Young's inequality}
\begin{equation}ab\le\frac{a^p}{p}+\frac{b^q}{q}.\tag{4.4}\end{equation}
\end{res}

\begin{res}{Lemma}{4.2}
Any two vectors $\boldsymbol v,\boldsymbol w\in\mathbb F^n$ satisfy \textbf{H\"older's inequality}
\begin{equation}\sum_{i=1}^n\left|v^i\right|\left|w^i\right|\le\|\boldsymbol v\|_p\|\boldsymbol w\|_q\tag{4.5}\end{equation}
\end{res}

\begin{res}{Proposition}{4.3}
Any two vectors $\boldsymbol v,\boldsymbol w\in\mathbb F^n$ satisfy \textbf{Minkowski's inequality}
\begin{equation}\|\boldsymbol v+\boldsymbol w\|_p\le\|\boldsymbol v\|_p+\|\boldsymbol w\|_p.\tag{4.6}\end{equation}
\end{res}

\begin{res}{Proposition}{4.4}
A normed vector space $(V,\|\cdot\|)$ becomes a metric space using the definition $d_V:V\times V\to\mathbb R:(v,w)\mapsto\|v-w\|$.
\end{res}

\begin{res}{Proposition}{4.5}
The norm itself is a continuous function $V\to\mathbb R$.
\end{res}

\begin{res}{Proposition}{4.6}
Vector addition is a continuous map from $V\times V$ to $V$, and scalar multiplication is a continuous map from $\mathbb F\times V$ to $V$.
\end{res}

\begin{res}{Proposition}{4.7}
Two norms $\|v\|_a$ and $\|v\|_b$ on $V$ are equivalent if and only if there exist constants $c,C>0$ such that
\begin{equation}\forall v\in V:c\,\|v\|_a\le\|v\|_b\le C\,\|v\|_a.\tag{4.10}\end{equation}
\end{res}

\begin{res}{Proposition}{4.8}
Let $V$ be a finite-dimensional vector space. Any two norms on $V$ are equivalent.
\end{res}

\begin{res}{Corollary}{4.9}
A consequence is that on any finite-dimensional vector space $V$, any norm is continuous with respect to the metric generated by any other norm.
\end{res}

\begin{res}{Proposition}{4.10}
For a metric space $(X,d_X)$, any sequence $(x_n\in X)_{n\in\mathbb N_0}$ that converges to a limit $x$ is a Cauchy sequence.
\end{res}

\begin{res}{Proposition}{4.11}
A normed space $(V,\|\cdot\|)$ is metric complete (and thus a Banach space) if and only if every absolutely convergent series converges.
\end{res}

\begin{res}{Proposition}{4.12}
For a linear map $\hat A\in\operatorname{Hom}(V,W)$, the following are equivalent:
\begin{enumerate}
\item $\hat A$ is bounded
\item $\hat A$ is continuous at the origin (or some other point).
\item $\hat A$ is continuous everywhere, and is in fact uniformly continuous.
\end{enumerate}
\end{res}

\begin{res}{Proposition}{4.13}
Let $(V,\|\cdot\|_V)$ and $(W,\|\cdot\|_W)$ be normed vector spaces. If $V$ is finite-dimensional, then $\mathcal B(V,W)=\operatorname{Hom}(V,W)$.
\end{res}

\begin{res}{Proposition}{4.14}
$(V,\|\cdot\|_V)$ be a normed vector space and $(W,\|\cdot\|_W)$ be a Banach space. The space of bounded linear maps $\mathcal B(V,W)$ together with the induced norm $\|\cdot\|_{V\to W}$ is a Banach space.
\end{res}

\begin{res}{Proposition}{4.15}
Let $(V,\|\cdot\|)$ be a normed vector space. The associated operator norm $\|\cdot\|_{V\to V}$ on $\operatorname{End}(V)$ is submultiplicative.
\end{res}

\begin{res}{Proposition}{4.16}
The norms $\|\mathsf A\|_p=\|\mathsf A\|_{p\to p}$ that are induced by the vector $p$-norm can be defined for any matrix size, and form a consistent family of matrix norms.
\end{res}

\begin{res}{Proposition}{4.17}
The Frobenius norm $\mathsf A_{\mathrm F}$ is consistent, and thus submultiplative and subordinate with respect to the vector 2-norm, to which it reduces for column matrices in $\mathbb F^{n\times1}$.
\end{res}

\begin{res}{Proposition}{4.18}
Let $\|\cdot\|$ be a submultiplicative norm on $\operatorname{End}(V)$ for a finite-dimensional vector space $V$. For any $\hat A\in\operatorname{End}(V)$, we find $\|\hat A\|\ge\rho_{\hat A}$.
\end{res}

\begin{res}{Proposition}{4.19}
Let $\|\cdot\|$ be a submultiplicative norm on $\operatorname{End}(V)$ for a finite-dimensional vector space $V$. For any $\hat A\in\operatorname{End}(V)$, we have
\begin{equation}\lim_{n\to\infty}\|\hat A^n\|^{1/n}=\rho_{\hat A}.\tag{4.24}\end{equation}
This result is known as the \textbf{Gelfand formula}.
\end{res}

\section{Inner Products and Orthogonality}

\begin{res}{Proposition}{5.1}
If the sesquilinear form $C$ is positive semidefinite but not positive definite, it is degenerate.
\end{res}

\begin{res}{Theorem}{5.2}
In an inner product space $(V,\langle\cdot,\cdot\rangle)$, all vectors $v,w\in V$ satisfy the \textbf{Cauchy-Schwarz-(Bunjakowski) inequality}
\begin{equation}|\langle v,w\rangle|^2\le\langle v,v\rangle\langle w,w\rangle\tag{5.9}\end{equation}
The inequality becomes an equality if and only if $v$ and $w$ are linearly dependent.
\end{res}

\begin{res}{Proposition}{5.3}
An inner product space $(V,\langle\ ,\ \rangle)$ is a normed vector space using the associated \textit{canonical norm} $\|v\|=\sqrt{\langle v,v\rangle}$.
\end{res}

\begin{res}{Proposition}{5.4}
The inner product $\langle\ ,\ \rangle:V\times V\to\mathbb F$ is continuous with respect to the norm it induces.
\end{res}

\begin{res}{Proposition}{5.5}
Consider an inner product space $(V,\langle\ ,\ \rangle)$. The associated norm satisfies the \textbf{parallelogram law}
\begin{equation}\|v+w\|^2+\|v-w\|^2=2(\|v\|^2+\|w\|^2),\tag{5.18}\end{equation}
the name of which clearly originates from a geometric property satisfied by two-dimensional parallellogramma in the two-dimensional Euclidean plane (see Figure 5.1).
\end{res}

\begin{res}{Theorem}{5.6}
Let $(V,\|\cdot\|)$ be a normed vector space such that the norm satisfies the parallelogram law in Eq.~(5.18). Then $V$ becomes an inner product space where the inner product induces the norm $\|\cdot\|$ by defining the inner product as
\begin{equation}\langle v,w\rangle=\begin{cases}\tfrac14\|v+w\|^2-\tfrac14\|v-w\|^2,&V\text{ real}\\[2pt]\tfrac14\|v+w\|^2-\tfrac14\|v-w\|^2+\tfrac{\mathrm i}4\|v-\mathrm iw\|^2-\tfrac{\mathrm i}4\|v+\mathrm iw\|^2,&V\text{ complex}\end{cases}\tag{5.19}\end{equation}
which is known as the \textbf{polarisation identity}.
\end{res}

\begin{res}{Proposition}{5.7}
A set of orthogonal vectors is linearly independent.
\end{res}

\begin{res}{Corollary}{5.8}
In a finite-dimensional vector space $V$, we can construct at most $n=\dim(V)$ orthogonal vectors.
\end{res}

\begin{res}[Pythagoras]{Theorem}{5.9}
Given a finite set of orthogonal vectors $\{v_i;i=1,\dots,n\}\subseteq V$, then
\begin{equation}\left\|\sum_{i=1}^nv_i\right\|^2=\sum_{i=1}^n\|v_i\|^2.\tag{5.21}\end{equation}
\end{res}

\begin{res}{Proposition}{5.10}
The orthogonal complement $S^\perp$ is a closed subspace of $V$.
\end{res}

\begin{res}{Proposition}{5.11}
Given a subset $S\subseteq V$, it holds that $S^\perp=(\mathbb FS)^\perp=(\overline{\mathbb FS})^\perp$.
\end{res}

\begin{res}{Proposition}{5.12}
Given a subset $S\subseteq V$. The intersection $S\cap S^\perp$ is $\{o\}$ if the zero vector is contained in $S$, and is empty otherwise.
\end{res}

\begin{res}{Theorem}{5.13} \textbf{(Projection)}
Consider a closed subspace $W\preccurlyeq V$ of a Hilbert space $V$.
\begin{itemize}
\item For any vector $v\in V$, there exists a unique closest vector $w\in W$ such that
\begin{equation}\|v-w\|=\inf_{w'\in W}\|v-w'\|.\tag{5.23}\end{equation}
\item The vector $w\in W$ closest to $v$ is the unique vector in $W$ such that $v-w\in W^\perp$.
\end{itemize}
The vector $w\in W$ is known as the \textbf{orthogonal projection} of $v$ onto $W$.
\end{res}

\begin{res}{Corollary}{5.14}
Let $W$ be a closed subspace of a Hilbert space $V$, then $V=W\oplus W^\perp$ and this \textbf{orthogonal direct sum} decomposition is unique.
\end{res}

\begin{res}{Lemma}{5.15}
Any nonzero orthogonal projector $\hat P$ on a Hilbert space $V$ has operator norm $\|P\|=1$. It is thus bounded and continuous.
\end{res}

\begin{res}{Proposition}{5.16}
Consider a Hilbert space $V$. Let $W$ be a closed subspace, so that $V=W\oplus W^\perp$, then the associated projection operator $\hat P_W$ is an orthogonal projector as characterised in Definition 5.19. Vice versa, any orthogonal projector gives rise to an orthogonal direct sum decomposition $V=\operatorname{im}(\hat P)\oplus\ker(\hat P)$, where $\operatorname{im}(\hat P)=\ker(\hat P)^\perp$ and both are closed subspaces.
\end{res}

\begin{res}{Proposition}{5.17}
For any subset $S$ of a Hilbert space $V$, it holds that $V=S^\perp\oplus S^{\perp\perp}$ where $S^{\perp\perp}=(S^\perp)^\perp$.
\end{res}

\begin{res}{Proposition}{5.18}
For any closed subspace $W$ of a Hilbert space $V$, $W^{\perp\perp}=W$.
\end{res}

\begin{res}{Proposition}{5.19}
For any subset $S$ of a Hilbert space $V$, it holds that $S^{\perp\perp}=\overline{\mathbb FS}$.
\end{res}

\begin{res}{Proposition}{5.20}
If $S$ is a complete set of a Hilbert space $V$, in the sense that the span of $S$ is dense in $V$, or thus $\overline{\mathbb FS}=V$, then $S^\perp=\{o\}$.
\end{res}

\begin{res}{Lemma}{5.21}
Consider an infinite-dimensional Hilbert space $V$ and an orthonormal sequence $(e_i)_{i\in\mathbb N_0}$. The orthogonal projection of a given vector $v\in V$ onto the subspace $W_n=\mathbb F\{e_i;i=1,\dots,n\}$ is given by $v_n=\sum_{i=1}^ne_i\langle e_i,v\rangle$.
\end{res}

\begin{res}{Lemma}{5.22}
For an infinite-dimensional Hilbert space $V$ and an orthonormal sequence $(e_i)_{i\in\mathbb N_0}$, \textbf{Bessel's inequality} states that for any $n\in\mathbb N_0$ and any $v\in V$
\begin{equation}\sum_{i=1}^n|\langle e_i,v\rangle|^2\le\|v\|^2.\tag{5.25}\end{equation}
so that in particular the series $\sum_{i=1}^{+\infty}|\langle e_i,v\rangle|^2$ converges to a value upper bounded by $\|v\|^2$.
\end{res}

\begin{res}[Expansion theorem]{Theorem}{5.23}
Let $V$ be a Hilbert space and $S=(e_i)_{i\in\mathbb N_0}$ a complete orthonormal sequence. Then $(e_i)_{i\in\mathbb N_0}$ is a Schauder basis and any vector $v\in V$ can be expanded as
\begin{equation}v=\sum_{i=1}^{+\infty}\langle e_i,v\rangle\,e_i.\tag{5.26}\end{equation}
\end{res}

\begin{res}{Corollary}{5.24}
Bessel's inequality becomes an equality
\begin{equation}\|v\|^2=\langle v,v\rangle=\sum_{i=1}^{+\infty}|\langle e_i,v\rangle|^2=\sum_{i=1}^{+\infty}\langle v,e_i\rangle\langle e_i,v\rangle\tag{5.28}\end{equation}
known as \textbf{Plancherel's identity}.
\end{res}

\begin{res}{Proposition}{5.25}
Let $V$ be a Hilbert space and $S=(e_i)_{i\in\mathbb N_0}$ a complete orthonormal sequence. It holds that
\begin{equation}\langle v,w\rangle=\sum_{i=1}^{+\infty}\langle v,e_i\rangle\langle e_i,w\rangle\tag{5.29}\end{equation}
which is known as \textbf{Parseval's identity}.
\end{res}

\begin{res}{Proposition}{5.26}
On a Hilbert space $V$, the inner product structure defines a canonical injective antilinear map $V\to V^*:v\mapsto\chi_v$, defined via the action
\[\chi_v(w)=\langle v,w\rangle,\qquad\forall w\in V.\]
Furthermore, the resulting linear functional $\chi_v$ is continuous and thus bounded.
\end{res}

\begin{res}[Riesz representation theorem]{Theorem}{5.27}
Every bounded linear functional $\xi$ on Hilbert space $V$ is uniquely associated with a vector $v_\xi\in V$ such that $\xi[w]=\langle v_\xi,w\rangle$ for all $w\in W$.
\end{res}

\begin{res}{Proposition}{5.28}
For every map $\hat A\in\mathcal B(V,W)$, we can construct a map in $\mathcal B(W,V)$, denoted as $\hat A^\dagger$ and referred to as the \textbf{adjoint} or \textbf{dagger} of $\hat A$, such that for any $v\in V$, $w\in W$,
\begin{equation}\langle w,\hat Av\rangle_W=\left\langle\hat A^\dagger w,v\right\rangle_V.\tag{5.40}\end{equation}
It furthermore holds that $\|\hat A\|=\|\hat A^\dagger\|$.
\end{res}

\begin{res}{Proposition}{5.29}
Let $\hat A$ be a bounded linear map between Hilbert spaces $V$ and $W$, then
\begin{equation}\left\|\hat A^\dagger\hat A\right\|=\left\|\hat A\hat A^\dagger\right\|=\left\|\hat A\right\|^2.\tag{5.44}\end{equation}
\end{res}

\begin{res}{Proposition}{5.30}
Let $\hat A$ be a bounded linear map between Hilbert spaces $V$ and $W$. We have $\ker(\hat A^\dagger)=\operatorname{im}(\hat A)^\perp$ and thus
\begin{equation}W=\ker(\hat A^\dagger)\oplus\overline{\operatorname{im}(\hat A)},\qquad V=\ker(\hat A)\oplus\overline{\operatorname{im}(\hat A^\dagger)}\tag{5.45}\end{equation}
\end{res}

\begin{res}{Proposition}{5.31}
A linear map $\hat Q:V\to W$ between Hilbert spaces $V$ and $W$ is isometric with respect to $d_W(w',w)=\|w'-w\|_W$ and $d_V(v',v)=\|v-v'\|_V$ if and only if
\begin{equation}\hat Q^\dagger\hat Q=\hat 1_V\tag{5.48}\end{equation}
Such a map is said to be a (linear) \textbf{isometry}.
\end{res}

\begin{res}{Proposition}{5.32}
An isometry $\hat U:V\to W$ that is furthermore surjective, and thus invertible, has $\hat U^\dagger$ as its inverse, and thus satisfies
\begin{equation}\hat U^\dagger\hat U=\hat 1_V,\qquad\hat U\hat U^\dagger=\hat 1_W.\tag{5.49}\end{equation}
Such a linear map is said to be \textbf{unitary} or, when $\mathbb F=\mathbb R$, \textbf{orthogonal}.
\end{res}

\begin{res}{Proposition}{5.33}
An operator $\hat A$ on a Hilbert space $V$ is normal if and only if $\|\hat Av\|=\|\hat A^\dagger v\|$ for all $v\in V$.
\end{res}

\begin{res}{Proposition}{5.34}
Let $v$ be an eigenvector with eigenvalue $\lambda$ fo a normal operator $\hat A$ on $V$, then $\hat A^\dagger v=\overline\lambda v$, i.e.\ $\hat A$ and $\hat A^\dagger$ share eigenspaces (up to complex conjugation of the corresponding eigenvalue).
\end{res}

\begin{res}{Proposition}{5.35}
For a normal operator $\hat A$, eigenvectors $v_\lambda$ and $v_\mu$ associated with distinct eigenvalues $\lambda\ne\mu$ are orthogonal.
\end{res}

\begin{res}{Lemma}{5.36}
Using the operator norm, a normal operator $\hat A$ on a Hilbert space $V$ satisfies $\|\hat A^n\|=\|\hat A\|^n$ for all $n\in\mathbb N$.
\end{res}

\begin{res}{Proposition}{5.37}
For a normal operator $\hat A$, $\|\hat A\|=\rho_{\hat A}$, i.e.\ the operator norm equals the spectral radius.
\end{res}

\begin{res}{Proposition}{5.38}
A normal operator $\hat A$ cannot be nilpotent.
\end{res}

If different pairs $(W,\chi)$ and $(W',\chi')$ satisfy this definition, then there exists an isomorphism $\phi\in\operatorname{Hom}(W,W')$ such that $\chi'=\phi\circ\chi$, and thus $W\cong W'$.

\section{Unitary Similarity and Unitary Equivalence}

\begin{res}{Proposition}{7.1}
The inverse of the discrete Fourier transformation is given by
\begin{equation}f_j=\frac1{\sqrt n}\sum_{k=0}^{n-1}F_k\exp\left(+\mathrm i\frac{2\pi}{n}jk\right),\qquad\forall j=0,\dots,n-1.\tag{7.3}\end{equation}
\end{res}

\begin{res}{Proposition}{7.2}
A circulant matrix $\mathsf A\in\mathbb F^{n\times n}$ is diagonalised by the Fourier transform matrix $\mathsf U\in\mathbb F^{n\times n}$, i.e.\ $\mathsf{AU}=\mathsf{U\Lambda}$ with $\mathsf\Lambda=\operatorname{diag}(\lambda_0,\lambda_1,\dots,\lambda_{n-1})$, where the eigenvalues are given by
\begin{equation}\lambda_k=\sum_{j=0}^{n-1}f_j\omega^{kj}=\sum_{j=0}^{n-1}f_j\mathrm e^{-\mathrm i\frac{2\pi}{n}kj}.\tag{7.5}\end{equation}
\end{res}

\begin{res}[Schur triangulation]{Theorem}{7.3}
Any matrix $\mathsf A\in\mathbb F^{n\times n}$ admits a \textbf{Schur decomposition}, which takes the form
\begin{equation}\mathsf A=\mathsf{ZTZ}^{\mathsf H}\tag{7.6}\end{equation}
where $\mathsf Z\in\mathrm U(n)$ and $\mathsf T\in\mathbb F^{n\times n}$ is upper triangular.
\end{res}

\begin{res}{Proposition}{7.4}
If $\mathsf A$ is a normal matrix, then for any of its Schur decompositions, the upper triangular matrix $\mathsf T$ is diagonal. Put differently, $\mathsf A$ is diagonalised by a unitary matrix and the Schur and eigenvalue decomposition coincide.
\end{res}

\begin{res}{Proposition}{7.5}
Let $\mathsf A\in\mathbb F^{n\times n}$ be such that it has a unique eigenvalue $\lambda_1$ with largest magnitude, i.e.\ $|\lambda_1|=\rho_{\mathsf A}$ (the spectral radius) and all other eigenvalues of $\mathsf A$ satisfy $|\lambda_i|<\rho_{\mathsf A}$. Assume that the eigenvalue $\lambda_1$ is simple or semisimple, i.e.\ its geometric and algebraic multiplicity coincide and there are no non-trivial Jordan blocks associated with $\lambda_1$. Let $\mathsf P_1$ be the projector onto the eigenspace $V_{\lambda_1}$ along the direct sum of all other eigenspaces and generalised eigenspaces. For any vector $\boldsymbol v$ for which $\mathsf P_1\boldsymbol v\ne\boldsymbol 0$, the power method will converge with $\mu_\infty=\lim_{k\to\infty}\mu_k=\lambda_1$ and $\boldsymbol q_\infty=\lim_{k\to\infty}\boldsymbol q_k$ a normalised eigenvector in $V_{\lambda_1}$.
\end{res}

\begin{res}{Proposition}{7.6}
Given a matrix $\mathsf A\in\mathbb F^{m\times n}$, there exist unitary matrices $\mathsf U\in\mathrm U(m)$ and $\mathsf V\in\mathrm U(n)$ so that
\begin{equation}\mathsf A=\mathsf{USV}^{\mathsf H}\tag{7.12}\end{equation}
where $\mathsf S\in\mathbb F^{m\times n}$ has only nonzero elements on the diagonal, which can be made positive and ordered such that $S^1{}_1=\sigma_1\ge S^2{}_2=\sigma_2\ge\ldots\ge S^p{}_p=\sigma_p>0$ for some $p\le\min(m,n)$. The numbers $(\sigma_1,\dots,\sigma_p)$ are known as the \textbf{singular values} of $\mathsf A$ and Eq.~(7.12) is referred to as the (full) \textbf{singular value decomposition} of $\mathsf A$.
\end{res}

\begin{res}{Proposition}{7.7}
The singular value decomposition exposes the rank of a matrix, namely $\operatorname{rank}(\mathsf A)=p$, the number of (non-zero) singular values. Correspondingly, $\mathsf U_p$, the column of $\mathsf U$ corresponding to the non-zero singular values, constitute an orthonormal basis for $\operatorname{im}(\mathsf A)$, while the columns $k=p+1,\dots,n$ of $\mathsf V$ provide an orthonormal basis for $\ker(\mathsf A)$.
\end{res}

\begin{res}{Proposition}{7.8}
Let $\mathsf A\in\mathbb F^{m\times n}$ represent a linear map between the standard spaces $\mathbb F^n$ and $\mathbb F^m$, for which we use the Euclidean norm $\|\cdot\|_2$. The induced norm on $\mathsf A$ is given by $\|\mathsf A\|_2=\sigma_1$, the largest singular value of $\mathsf A$.
\end{res}

\begin{res}{Proposition}{7.9}
The Frobenius norm of $\mathsf A\in\mathbb F^{m\times n}$ is given by $\|\mathsf A\|_{\mathrm F}=\sqrt{\sum_{i=1}^p\sigma_i^2}$.
\end{res}

\begin{res}{Proposition}{7.10}
The condition number of an invertible matrix $\mathsf A\in\mathbb F^{n\times n}$ is given by $\kappa(\mathsf A)=\sigma_1/\sigma_n$.
\end{res}

\begin{res}{Proposition}{7.11}
Consider a general matrix $\mathsf A\in\mathbb F^{m\times n}$. The minimum-norm least-squares solution associated with the linear system $\mathsf A\boldsymbol x=\boldsymbol y$ is unique and given by $\boldsymbol x^*=\mathsf V_p\mathsf S_p^{-1}\mathsf U_p^{\mathsf H}\boldsymbol y$, using the compact singular value decomposition of $\mathsf A$.
\end{res}

\begin{res}{Theorem}{7.12}
Given $\mathsf A\in\mathbb F^{m\times n}$, the rank $r$ matrix $\mathsf B$ that minimises $\|\mathsf A-\mathsf B\|_2$ using the induced 2-norm is given by the truncated singular value decomposition $\mathsf B=\mathsf U_r\mathsf S_r\mathsf V_r^{\mathsf H}$ of $\mathsf A$.
\end{res}

\begin{res}{Theorem}{7.13}
Given $\mathsf A\in\mathbb F^{m\times n}$, the rank $r$ matrix $\mathsf B$ that minimises $\|\mathsf A-\mathsf B\|_{\mathrm F}$ using the Frobenius norm is given by the truncated singular value decomposition $\mathsf B=\mathsf U_r\mathsf S_r\mathsf V_r^{\mathsf H}$ of $\mathsf A$.
\end{res}

\begin{res}{Proposition}{7.14}
Consider a matrix $\mathsf A\in\mathbb F^{m\times n}$ with $m\ge n$. It can be decomposed as
\begin{equation}\mathsf A=\mathsf{UP}\tag{7.17}\end{equation}
where $\mathsf U\in\mathbb F^{m\times n}$ is isometric ($\mathsf U^{\mathsf H}\mathsf U=\mathsf I_n$) and $\mathsf P\in\mathbb F^{n\times n}$ is Hermitian and positive semidefinite. This decomposition is known as the \textbf{polar decomposition}.
\end{res}

\begin{res}{Proposition}{7.15}
Given $\mathsf A\in\mathbb F^{m\times n}$ with $m\ge n$, the isometric matrix $\mathsf B$ that minimises $\|\mathsf A-\mathsf B\|_{\mathrm F}$ is given by the isometric factor in the polar decomposition of $\mathsf A$.
\end{res}

\section{Function Spaces}

\begin{res}{Proposition}{8.1}
The set $\mathcal L^p(I;\mathbb F)$ is actually a vector space (a subspace of $\mathbb F^I$).
\end{res}

\begin{res}{Proposition}{8.2}
On the space $\mathcal L^p(I;\mathbb F)$, the prescription $f\mapsto\|f\|_p$ given in Eq.~(8.1) is a seminorm.
\end{res}

\begin{res}[Riesz-Fischer]{Theorem}{8.3}
The Lebesgue spaces $L^p(I)$ for $p\ge1$ are metric complete, i.e.\ they are Banach spaces.
\end{res}

\begin{res}{Proposition}{8.4}
The space $L^2(I)$ admits an inner product given by
\begin{equation}\langle f,g\rangle=\int_I\overline{f(x)}g(x)\,\mathrm dx\tag{8.12}\end{equation}
which is such that $\langle f,f\rangle=(\|f\|_2)^2$.
\end{res}

\begin{res}{Proposition}{8.5}
The set of orthogonal polynomials $\{p_n;n\in\mathbb N\}$ constructed on $L^2_w(I)$ is such that every $p_n(x)$ is real-valued polynomial of degree $n$, that is orthogonal to any polynomial of lower degree.
\end{res}

\begin{res}{Proposition}{8.6}
The set of orthogonal polynomials $\{p_n;n\in\mathbb N\}$ constructed on $L^2_w(I)$ is governed by a recurrence relation of the form
\begin{equation}xp_n(x)=a_np_{n+1}(x)+b_np_n(x)+c_np_{n-1}(x)\tag{8.15}\end{equation}
for all $n\in\mathbb N$, where clearly the last term is not present for $n=0$, i.e.\ $c_0=0$. It furthermore holds that $c_n\langle p_{n-1},p_{n-1}\rangle_w=a_{n-1}\langle p_n,p_n\rangle_w$, or thus $c_n=a_{n-1}$ if the different $p_n(x)$ are normalised to have an $n$-independent norm.
\end{res}

\begin{res}{Proposition}{8.7}
For the set of orthogonal polynomials $\{p_n;n\in\mathbb N\}$ constructed on $L^2_w(I)$, the orthogonal projector onto the first $n$ polynomials is given by
\begin{equation}(\hat P_nf)(x)=\int_Iw(y)K_n(x,y)f(y)\,\mathrm dy=\int_Iw(y)\left[\sum_{k=0}^n\frac{p_k(x)p_k(y)}{\langle p_k,p_k\rangle_w}\right]f(y)\,\mathrm dy\tag{8.16}\end{equation}
where $K(x,y)$ is given by the \textit{Christoffel-Darboux formula}
\begin{equation}\sum_{k=0}^n\frac{p_k(x)p_k(y)}{\langle p_k,p_k\rangle_w}=\begin{cases}\dfrac{a_n}{\langle p_n,p_n\rangle_w}\dfrac{p_{n+1}(x)p_n(y)-p_n(x)p_{n+1}(y)}{x-y},&x\ne y\\[8pt]\dfrac{a_n}{\langle p_n,p_n\rangle_w}\bigl(p'_{n+1}(x)p_n(y)-p'_n(x)p_{n+1}(y)\bigr),&x=y\end{cases}.\tag{8.17}\end{equation}
\end{res}

\begin{res}{Proposition}{8.8}
For the set of orthogonal polynomials $\{p_n;n\in\mathbb N\}$ constructed on $L^2_w(I)$, $p_n(x)$ has $n$ distinct roots on $I$. Furthermore, the zeros of $p_n$ and $p_{n+1}$ alternate, i.e.\ the roots of $p_n$ lie in between those of $p_{n+1}$.
\end{res}

\begin{res}{Proposition}{8.9}
The numerical approximation $\sum_{i=1}^na^if(x_i)$ to the integral $\int_Iw(x)f(x)\,\mathrm dx$ can be made exact for any $f(x)$ that is a polynomial of at most degree $n-1$
\end{res}

\begin{res}{Theorem}{8.10}
The numerical approximation $\sum_{i=1}^na^if(x_i)$ to the integral $\int_Iw(x)f(x)\,\mathrm dx$ can be made exact for any $f(x)$ that is a polynomial of at most degree $2n-1$, if the set of points $x_i$, $i=1,\dots,n$ is chosen to coincide with the roots of the degree $n$ orthogonal polynomial $p_n(x)$ constructed on $I$ with weight function $w(x)$.
\end{res}

\begin{res}{Proposition}{8.11}
On the Hilbert space $L^2([0,L])$, the set of Fourier modes
\begin{equation}S=\left\{\varphi_k(x)=\frac1{\sqrt L}\mathrm e^{+\mathrm i\frac{2\pi}{L}kx};\ k=-n,-n+1,\dots,0,\dots,n-1,n\right\}\tag{8.56}\end{equation}
constitutes an orthonormal basis for the space of trigonometric polynomials.
\end{res}

\begin{res}{Proposition}{8.12}
The orthogonal projection of a function $f\in L^2([0,L])$ onto the subspace of trigonometric polynomials of maximal degree $n$ given by
\begin{align}
f_n(x)=(\hat P_nf)(x)&=\sum_{k=-n}^{+n}\langle\varphi_k,f\rangle\varphi_k(x)=\sum_{k=-n}^{+n}\widehat f_k\varphi_k(x)=\frac1{\sqrt L}\sum_{k=-n}^{+n}\widehat f_k\mathrm e^{+\mathrm i\frac{2\pi}{L}kx}\tag{8.60}\\
&=\sum_{k=-n}^{+n}\frac1L\int_0^Lf(t)\mathrm e^{+\mathrm i\frac{2\pi}{L}k(x-t)}\,\mathrm dt=\int_0^LD_n(x-t)f(t)\,\mathrm dt\tag{8.61}
\end{align}
where we have introduced the \textbf{Dirichlet kernel}
\begin{equation}D_n(x)=\frac1L\sum_{k=-n}^{+n}\mathrm e^{+\mathrm i\frac{2\pi}{L}kx}=\frac{\sin\left(\frac{(2n+1)\pi}{L}x\right)}{L\sin\left(\frac\pi Lx\right)}.\tag{8.62}\end{equation}
\end{res}

\begin{res}{Proposition}{8.13}
The Fourier coefficients $\widehat f_k$ in Eq.~(8.59) are well defined for any function $f\in L^1([0,L])$, and satisfy
\begin{equation}\left|\widehat f_k\right|\le\frac1{\sqrt L}\int_0^L\left|f(x)\mathrm e^{-\mathrm i\frac{2\pi}{L}kx}\right|\mathrm dx=\frac1{\sqrt L}\int_0^L|f(x)|\,\mathrm dx=\frac{\|f\|_1}{\sqrt L},\quad\forall k\in\mathbb Z.\tag{8.65}\end{equation}
Denoting the double-sided sequence of Fourier coefficients as $\widehat f=(\widehat f_k)_{k\in\mathbb Z}$, we thus find $\left\|\widehat f\right\|_\infty=\sup_{k\in\mathbb Z}\left|\widehat f_k\right|<\infty$, i.e.\ $\widehat f\in\ell^\infty(\mathbb Z)$.
\end{res}

\begin{res}{Proposition}{8.14}
The map from a function $f\in L^1([0,L])$ to its sequence of Fourier coefficients $\widehat f\in\ell^\infty(\mathbb Z)$ has the following elementary properties:
\begin{enumerate}
\item \textbf{Linearity}:
\begin{equation}h(x)=af(x)+bg(x)\implies\widehat h_k=a\widehat f_k+b\widehat g_k,\quad\forall f,g\in L^1([0,L]),\forall a,b\in\mathbb C\tag{8.66}\end{equation}
\item \textbf{Translation} (shift in space/time):
\begin{equation}h(x)=f(x-x_0)\implies\widehat h_k=\mathrm e^{-\mathrm i\frac{2\pi}{L}kx_0}\widehat f_k,\quad\forall f\in L^1([0,L]),\forall x_0\in\mathbb R\tag{8.67}\end{equation}
\item \textbf{Modulation} (shift in frequency):
\begin{equation}h(x)=f(x)\mathrm e^{\mathrm i\frac{2\pi}{L}k_0x}\implies\widehat h_k=\widehat f_{k-k_0},\quad\forall f\in L^1([0,L]),\forall k_0\in\mathbb Z\tag{8.68}\end{equation}
\item \textbf{Conjugation}:
\begin{equation}h(x)=\overline{f(x)}\implies\widehat h_k=\overline{\widehat f_{-k}},\quad\forall f\in L^1([0,L])\tag{8.69}\end{equation}
\item \textbf{Time/frequency reversal}:
\begin{equation}h(x)=f(-x)\implies\widehat h_k=\widehat f_{-k},\quad\forall f\in L^1([0,L])\tag{8.70}\end{equation}
\item \textbf{Discrete scaling}:
\begin{equation}h(x)=f(sx)\implies\widehat h_k=\begin{cases}\frac1s\widehat f_{k/s},&k\text{ is a multiple of }s\\0,&\text{otherwise}\end{cases},\quad\forall f\in L^1([0,L]),\forall s\in\mathbb N_0\tag{8.71}\end{equation}
\end{enumerate}
In properties 2, 5 en 6, we furthermore assume that $f$ is extended periodically outside of $[0,L)$ in order to define $h$. For property 2, we could also have written $h(x)=f((x-x_0)\bmod L)$, which illustrates that this property is also periodic in $x_0$ with period $L$. For property 5, we could have defined $h(x)=f(L-x)$. For property 6, note the importance of the scale factor $s$ to be an integer, in order for $h(x)=f(sx)$ to also be a periodic function with period $L$ (although $h$ is actually periodic with smallest period $L/s$).
\end{res}

\begin{res}{Proposition}{8.15}
The (periodic) convolution $f*g$ of two functions $f,g\in L^1([0,L])$ is again a function in $L^1([0,L])$.
\end{res}

\begin{res}{Proposition}{8.16}
For functions $f,g\in L^1([0,L])$ and $h=f*g$, we find
\begin{equation}h(x)=(f*g)(x)=\int_0^Lf(x-y\bmod L)g(y)\,\mathrm dy\implies\widehat h_k=\sqrt L\,\widehat f_k\widehat g_k.\tag{8.73}\end{equation}
\end{res}

\begin{res}{Theorem}{8.17}
Consider a function $f\in L^1([0,L])$ and $f_n$ its approximation as a trigonometric polynomial in Eq.~(8.61). It holds that
\begin{equation}f=\lim_{n\to\infty}\frac{f_0+f_1+\ldots+f_n}{n+1}=\lim_{n\to\infty}\tilde f_n=\lim_{n\to\infty}\hat Q_nf\tag{8.75}\end{equation}
in 1-norm, i.e.\ $\lim_{n\to\infty}\left\|\tilde f_n-f\right\|_1=0$.
\end{res}

\begin{res}[Riemann-Lebesgue]{Lemma}{8.18}
The Fourier coefficients of a function $f\in L^1([0,L])$ satisfy $\lim_{k\to\pm\infty}\widehat f_k=0$
\end{res}

\begin{res}{Theorem}{8.19}
Consider a function $f\in L^2([0,L])$. The associated Fourier series satisfies
\begin{equation}f=\lim_{n\to\infty}f_n=\lim_{n\to\infty}\frac1{\sqrt L}\sum_{k=-n}^{+n}\widehat f_k\mathrm e^{+\mathrm i\frac{2\pi}{L}kx}\tag{8.76}\end{equation}
in 2-norm, i.e.\ $\lim_{n\to\infty}\|f_n-f\|_2=0$. The map between $f$ and its sequence of Fourier components is a unitary map from $L^2([0,L])$ to $\ell^2(\mathbb Z)$, with thus in particular
\begin{equation}\|f\|_2=\int_0^L|f(x)|^2\,\mathrm dx=\sum_{k\in\mathbb Z}\left|\widehat f_k\right|^2=\left\|\widehat f\right\|_2.\tag{8.77}\end{equation}
\end{res}

\begin{res}{Proposition}{8.20}
For $f\in C^p(\mathbb T^1_L)$, i.e.\ $f$ and all of its derivatives up to $f^{(p)}$ are continuous (and thus square integrable) on the circle of circumference $L$, we find
\begin{equation}h(x)=f^{(p)}(x)\implies\widehat h_k=\left(\mathrm i\frac{2\pi}{L}k\right)^p\widehat f_k.\tag{8.80}\end{equation}
\end{res}

\begin{res}{Proposition}{8.21}
If $f\in H^p([0,L])$ with $p>1/2$, which means that $\sum_{k\in\mathbb Z}|k|^{2p}\left|\widehat f_k\right|^2$ is finite, then the sequence of partial sums of the Fourier series of $f$ converges uniformly, and it holds that there is a constant $C_p$ such that
\begin{equation}\|f_n-f\|_\infty\le\frac{C_p}{n^{p-1/2}}\left[\sum_{k\in\mathbb Z}|k|^{2p}\left|\widehat f_k\right|^2\right]^{1/2}.\tag{8.84}\end{equation}
\end{res}

\begin{res}{Proposition}{8.22}
For $f,g\in L^2([0,L])$, the convolution $h=f*g$ is a continuous function with uniformly converging Fourier series.
\end{res}

\begin{res}{Proposition}{8.23}
For functions $f,g\in C(\mathbb T_L)$ which are such that $\widehat f,\widehat g\in\ell^1(\mathbb Z)$, we find
\begin{equation}h(x)=f(x)g(x)\implies\widehat h_k=\frac1{\sqrt L}(\widehat f*\widehat g)_k=\frac1{\sqrt L}\sum_{\ell\in\mathbb Z}\widehat f_{k-l}\widehat g_l.\tag{8.86}\end{equation}
\end{res}

\begin{res}{Proposition}{8.24}
An integral operator on $L^2(I)$ with kernel $A:I\times I\to\mathbb C$ is bounded if $\int_I\int_I|A(x,y)|^2\,\mathrm dy\mathrm dx$ is finite.
\end{res}

\begin{res}{Proposition}{8.25}
The multiplication operator $\hat M_h=h(\hat X)$ is bounded if $\operatorname*{ess\,sup}_{x\in I}|h(x)|$ is finite.
\end{res}

\begin{res}{Proposition}{8.26}
Differential operators on $L^2(I)$ are always unbounded.
\end{res}

\begin{res}{Proposition}{8.27}
Consider two bounded operators $\hat A$ and $\hat B$ on a Hilbert space $H$; the commutator $\hat C=\hat A\hat B-\hat B\hat A=[\hat A,\hat B]$ cannot equal the identity operator.
\end{res}

\begin{res}{Proposition}{8.28}
For a densely defined unbounded operator $\hat A$ on a Hilbert space $A$ with domain $\mathcal D_{\hat A}$, there exits an adjoint operator $\hat A^\dagger$ such that
\begin{equation}\langle w,\hat Av\rangle=\langle\hat A^\dagger w,v\rangle,\quad\forall v\in\mathcal D_{\hat A},w\in\mathcal D_{\hat A^\dagger}.\tag{8.105}\end{equation}
\end{res}

\begin{res}{Proposition}{8.29}
For two densely defined operators $\hat A$, $\hat B$ on a Hilbert space $H$, it holds that
\begin{equation}\hat A\subset\hat B\implies\hat B^\dagger\subset\hat A^\dagger.\tag{8.106}\end{equation}
\end{res}

\begin{res}{Proposition}{8.30}
Consider a densely defined operator $\hat A$ on $H$, for which also $\hat A^\dagger$ is densely defined. We can then also define $\hat A^{\dagger\dagger}$, and it holds that $\hat A\subset\hat A^{\dagger\dagger}$.
\end{res}

\begin{res}{Proposition}{8.31}
If a densely defined operator $\hat A$ is symmetric, then $\hat A^\dagger$ is densely defined and it holds that
\begin{equation}\hat A\subset\hat A^{\dagger\dagger}\subset\hat A^\dagger.\tag{8.108}\end{equation}
\end{res}

\begin{res}{Proposition}{8.32}
A linear combination of compact operators is compact.
\end{res}

\begin{res}{Proposition}{8.33}
The composition of a compact operator with a bounded operator is compact.
\end{res}

\begin{res}{Proposition}{8.34}
The adjoint of a compact operator is compact.
\end{res}

\begin{res}{Proposition}{8.35}
If a sequence $(\hat A_n)_{n\in\mathbb N_0}$ of compact operators $\hat A_n$ has a limit $\lim_{n\to\infty}\hat A_n=\hat A$ (with respect to the induced operator norm), then $\hat A$ is compact.
\end{res}

\begin{res}{Proposition}{8.36}
Any Hilbert-Schmidt operator is compact.
\end{res}

\begin{res}{Lemma}{8.37}
A self-adjoint compact operator necessarily has a real eigenvalue $\lambda$ equal to plus or minus $\|A\|$.
\end{res}

\begin{res}{Theorem}{8.38}
A compact self-adjoint operator admits a spectral decomposition
\begin{equation}\hat A=\sum_n\lambda_n\hat P_{\lambda_n}\tag{8.116}\end{equation}
which provides a direct sum decomposition of the Hilbert space
\begin{equation}H=V_0\bigoplus_nV_{\lambda_n}.\tag{8.117}\end{equation}
Here, $V_0=\ker(\hat A)$ is the kernel of $\hat A$ (which can be infinite dimensional). We have not denoted the summation limits of $n$ which labels the non-zero eigenvalues $\lambda_n$, as this can be an infinite (but countable) or finite range. It furthermore holds that $|\lambda_n|$ is non-increasing and, when it is an infinite sequence, $\lim_{n\to\infty}|\lambda_n|=0$.
\end{res}

\begin{res}{Theorem}{8.39}
For a compact operator $\hat A$, there exists a non-increasing sequence $(\sigma_n)_{n\in\mathbb N_0}$ of nonnegative numbers $\sigma_n\ge0$ with associated vectors $u_n$ and $v_n$ satisfying
\begin{equation}\hat Au_n=\sigma_nv_n,\qquad\hat A^\dagger v_n=\sigma_nu_n,\tag{8.118}\end{equation}
so that the action of $\hat A$ on a general vector can then be written as
\begin{equation}\hat Aw=\sum_n\sigma_nu_n\langle v_n,w\rangle.\tag{8.119}\end{equation}
Furthermore, if the number of terms is infinite, then $\lim_{n\to\infty}\sigma_n=0$. This representation is known as the \textbf{Schmidt decomposition}.
\end{res}

\begin{res}{Proposition}{8.40}
The resolvent set is an open set.
\end{res}

\begin{res}{Proposition}{8.41}
For a self-adjoint operator $\hat A$, it holds that $\sigma^{(\mathrm r)}_{\hat A}=\emptyset$ and $\sigma_{\hat A}=\sigma^{(\mathrm p)}_{\hat A}\cup\sigma^{(\mathrm c)}_{\hat A}\subseteq\mathbb R$.
\end{res}

\begin{res}{Proposition}{8.42}
For a self-adjoint operator $\hat A$ on a separable Hilbert space $H$, the point spectrum $\sigma^{(\mathrm p)}_{\hat A}$ must be countable.
\end{res}

\section{Fourier Calculus and Distributions}

\begin{res}{Proposition}{9.1}
The map from a function $f\in L^1([0,L])$ to its Fourier transform $\widehat f\in C_{\mathrm b}(\mathbb R)$ has the following elementary properties:
\begin{enumerate}
\item \textbf{Linearity}:
\begin{equation}h(x)=af(x)+bg(x)\implies\widehat h(\xi)=a\widehat f(\xi)+b\widehat g(\xi),\quad\forall f,g\in L^1(\mathbb R),\forall a,b\in\mathbb C\tag{9.9}\end{equation}
\item \textbf{Translation} (shift in space/time):
\begin{equation}h(x)=f(x-x_0)\implies\widehat h(\xi)=\mathrm e^{-\mathrm i2\pi\xi x_0}\widehat f(\xi),\quad\forall f\in L^1(\mathbb R),\forall x_0\in\mathbb R\tag{9.10}\end{equation}
\item \textbf{Modulation} (shift in frequency):
\begin{equation}h(x)=f(x)\mathrm e^{\mathrm i2\pi\xi_0x}\implies\widehat h(\xi)=\widehat f(\xi-\xi_0),\quad\forall f\in L^1(\mathbb R),\forall\xi_0\in\mathbb R\tag{9.11}\end{equation}
\item \textbf{Conjugation}:
\begin{equation}h(x)=\overline{f(x)}\implies\widehat h(\xi)=\overline{\widehat f(-\xi)},\quad\forall f\in L^1(\mathbb R)\tag{9.12}\end{equation}
\item \textbf{Time/frequency reversal}:
\begin{equation}h(x)=f(-x)\implies\widehat h(\xi)=\widehat f(-\xi),\quad\forall f\in L^1(\mathbb R)\tag{9.13}\end{equation}
\item \textbf{Scaling}:
\begin{equation}h(x)=f(sx)\implies\widehat h(\xi)=\frac1s\widehat f(\xi/s)\quad\forall f\in L^1(\mathbb R),\forall s\in\mathbb R_{>0}\tag{9.14}\end{equation}
\end{enumerate}
The last two properties could be combined in $h(x)=f(sx)\implies\widehat h(\xi)=\frac1{|s|}\widehat f(\xi/s)$, $\forall s\in\mathbb R_{\ne0}$.
\end{res}

\begin{res}{Theorem}{9.2}
For any function $f\in L^1(\mathbb R)$, the Fourier transform $\widehat f(\xi)$ is
\begin{itemize}
\item continuous: $\widehat f(\xi)\in C^0(\mathbb R)$;
\item bounded: $\left\|\widehat f\right\|_\infty=\sup_\xi\left|\widehat f(\xi)\right|\le\|f\|_1$;
\item vanishing at infinity: $\lim_{\xi\to\pm\infty}\left|\widehat f(\xi)\right|=0$ (known as the Riemann-Lebesgue lemma).
\end{itemize}
This is written as $\widehat f\in C^0_0(\mathbb R)$, the space of continuous functions that vanish at infinity.
\end{res}

\begin{res}[Convolution]{Theorem}{9.3}
For functions $f,g\in L^1(\mathbb R)$ and $h=f*g$, we find
\begin{equation}\widehat h(\xi)=\widehat f(\xi)\widehat g(\xi).\tag{9.16}\end{equation}
\end{res}

\begin{res}{Proposition}{9.4}
For a continuous differentiable function $f$ with a piecewise continuous derivative $g$ such that $f,g=f'\in L^1(\mathbb R)$ and $\lim_{x\to\pm\infty}|f(x)|=0$, it holds that\footnote{The condition that $f$ goes to zero at infinite might seem redundant. It is certainly not implied by $f\in L^1(\mathbb R)$, you could easily imagine a function that takes the value 1 in a small region of width $2^{-|n|}$ for any integer $n\in\mathbb Z$, for which the total integral is well defined. If $f$ needs to be differentiable (and thus continuous) and furthermore have a derivative $f'\in L^1(\mathbb R)$, that's already more stringent, but to actually be a sufficient condition to prove $\lim_{x\to\pm\infty}|f(x)|=0$, $f$ needs to be absolutely continuous.}
\[\widehat g(\xi)=\mathrm i2\pi\xi\widehat f(\xi)\]
\end{res}

\begin{res}{Lemma}{9.5}
For a continuous function $f$ supported on a compact interval $I\subseteq\mathbb R$, it holds that $\|f\|_2=\left\|\widehat f\right\|_2$ and thus $\widehat f\in L^2(\mathbb R)$.
\end{res}

\begin{res}[Plancherel]{Theorem}{9.6}
For all $f\in L^2(\mathbb R)$ it holds that $\|f\|_2=\|\hat F(f)\|_2=\left\|\widehat f\right\|_2$.
\end{res}

\begin{res}[Parseval]{Theorem}{9.7}
For all $f,g\in L^2(\mathbb R)$, it holds that $\langle f,g\rangle=\langle\hat F(f),\hat F(g)\rangle=\left\langle\widehat f,\widehat g\right\rangle$.
\end{res}

\begin{res}[Weak Parseval's relation]{Theorem}{9.8}
For all $f,g\in L^2(\mathbb R)$, it holds that
\begin{equation}\int_{-\infty}^{+\infty}f(x)\widehat g(x)\,\mathrm dx=\int_{-\infty}^{+\infty}\widehat f(x)g(x)\,\mathrm dx\tag{9.20}\end{equation}
\end{res}

\begin{res}{Corollary}{9.9}
The adjoint of $\hat F$ is given by $\hat F^\dagger(f)=\overline{\hat F(\overline f)}$, or thus, by
\begin{equation}(\hat F^\dagger f)(x)=\lim_{n\to\infty}\int_{-n}^{+n}f(\xi)\mathrm e^{+\mathrm i2\pi\xi x}\,\mathrm d\xi.\tag{9.21}\end{equation}
\end{res}

\begin{res}{Theorem}{9.10}
The Plancherel-Fourier operator $\hat F$ defined in Definition 9.4 is a unitary transformation from $L^2(\mathbb R)$ onto itself, and for all $f\in L^2(\mathbb R)$ and $\widehat f=\hat F(f)$, we have the relations
\begin{align}
\lim_{n\to\infty}\left\|\widehat f(\xi)-\int_{-n}^{+n}f(x)\mathrm e^{-\mathrm i2\pi\xi x}\,\mathrm dx\right\|_2&=0\tag{9.22}\\
\lim_{n\to\infty}\left\|f(x)-\int_{-n}^{+n}f(x)\mathrm e^{+\mathrm i2\pi\xi x}\,\mathrm d\xi\right\|_2&=0\tag{9.23}
\end{align}
If $f$ is absolutely integrable ($f\in L^1(\mathbb R)\cap L^2(\mathbb R)$) then $\widehat f$ is continuous and the first relation holds pointwise, whereas the second relation holds pointwise if $\widehat f$ is absolutely integrable (which then implies that $f\in C_0(\mathbb R)$).
\end{res}

\begin{res}{Proposition}{9.11}
For test functions $\varphi\in\mathcal S(\mathbb R)$, the Fourier transform satisfies $\widehat\varphi\in\mathcal S(\mathbb R)$.
\end{res}

\begin{res}{Proposition}{9.12}
Any function $f\in L^1_{\mathrm{loc}}(\Omega)$ defines a distribution $T_f$, known as the \textbf{regular distribution} associated with $f$, using the prescription
\begin{equation}T_f[\varphi]=\int_\Omega f(x)\varphi(x)\,\mathrm dx.\tag{9.40}\end{equation}
\end{res}

\begin{res}{Proposition}{9.13}
The product of a smooth function $\psi$ and a distribution $T$ satisfies the Leibniz rule
\begin{equation}(\psi T)'=\psi'T+\psi T'.\tag{9.56}\end{equation}
\end{res}

\begin{res}{Proposition}{9.14}
For a converging sequence of distributions $(T_n)_{n\in\mathbb N_0}$ in $\mathcal D^*(\mathbb R)$ and a smooth function $\psi\in C^\infty(\mathbb R)$, it holds that
\begin{equation}\lim_{n\to\infty}T_n'=T',\qquad\lim_{n\to\infty}\psi T_n=\psi T.\tag{9.73}\end{equation}
\end{res}

\begin{res}[Sokhotski--Plemelj]{Theorem}{9.15}
The distributional limit of $\frac1{x\pm\mathrm is}$ for $s\to0^+$ is given by
\begin{equation}\lim_{s\to0^+}\frac1{x\pm\mathrm is}=\operatorname{Pv}\frac1x\mp\mathrm i\pi\delta(x).\tag{9.74}\end{equation}
\end{res}

\begin{res}{Proposition}{9.16}
Let $f\in L^1(\mathbb R^d)$ be a nonnegative function with
\begin{equation}\int_{\mathbb R^d}f(\boldsymbol x)\,\mathrm d^d\boldsymbol x=1.\tag{9.78}\end{equation}
Then $f_n=n^d\hat\sigma_nf$ or thus $f_n(\boldsymbol x)=n^df(n\boldsymbol x)$ defines a Dirac sequence, i.e.
\begin{equation}\lim_{n\to\infty}T_{f_n}[\varphi]=\lim_{n\to\infty}\int_{\mathbb R^d}n^df(n\boldsymbol x)\varphi(\boldsymbol x)\,\mathrm d^d\boldsymbol x=\varphi(\boldsymbol 0),\quad\forall\varphi\in\mathcal D(\mathbb R^d).\tag{9.79}\end{equation}
\end{res}

\begin{res}{Proposition}{9.17}
The sequence of functions
\begin{equation}f_n(x)=\frac1\pi\frac{\sin(nx)}{x}=\frac n\pi\operatorname{sinc}(nx)\tag{9.89}\end{equation}
with $\operatorname{sinc}(x)=\sin(x)/x$ is a Dirac sequence, i.e.\ $\lim_{n\to\infty}T_{f_n}=\delta$.
\end{res}

\begin{res}[Poisson summation formula]{Theorem}{9.18}
For a function $\varphi\in\mathcal S(\mathbb R)$, i.e.\ a function that is smooth and decaying faster than any polynomial, it holds that
\begin{equation}\sum_{n\in\mathbb Z}\varphi(n)=\sum_{k\in\mathbb Z}\widehat\varphi(k).\tag{9.110}\end{equation}
\end{res}

\begin{res}{Proposition}{9.19}
For a linear system $x=\hat Au$ with a translation invariant linear operator $\hat A$, the response $x(t)$ can be obtained as the convolution $u*h$, where $h=\hat A\delta$ is known as the \textbf{impulse response}.
\end{res}

\begin{res}{Proposition}{9.20}
Consider the samples $F_j=f(x_j)$ of the function $f$ on $\mathbb R$, for all $x_j=j\epsilon$ with $j\in\mathbb Z$ and $\epsilon$ a sampling period. The discrete-time Fourier transform of the samples $F=(F_j)_{j\in\mathbb Z}$ is related to the Fourier transform of $f$ as
\begin{equation}\widehat F(\chi)=\sum_{n\in\mathbb Z}\frac1\epsilon\widehat f\left(\frac\chi\epsilon-\frac n\epsilon\right).\tag{9.125}\end{equation}
\end{res}

\begin{res}{Corollary}{9.21}
When the function $f$ is bandwidth limited with frequency $\Xi$, then by choosing a sampling rate $1/\epsilon>2\Xi$, known as the \textbf{Nyquist rate}, the Fourier transform $\widehat f$ of $f$ can be perfectly reconstructed from the discrete-time Fourier transform $\widehat F$ of the samples $F_j=f(j\epsilon)$, namely as
\begin{equation}\widehat f(\xi)=\begin{cases}\epsilon\widehat F(\epsilon\xi),&|\xi|\le\frac1{2\epsilon}\\0,&|\xi|>\frac1{2\epsilon}\end{cases}.\tag{9.126}\end{equation}
\end{res}

\begin{res}{Proposition}{9.22}
Consider a function $f$ on $\mathbb R$ and samples $F_j=f(j\epsilon)$ with sampling period $\epsilon$. Define a reconstruction of $f$ from its samples via the \textbf{Whittaker--Shannon interpolation formula}
\begin{equation}f^{(\mathrm{rec})}(x)=\sum_{j\in\mathbb Z}F_j\operatorname{sinc}\left(\pi\frac{x-j\epsilon}{\epsilon}\right)\tag{9.127}\end{equation}
with $\operatorname{sinc}(x)=\sin(x)/x$. It follows that
\begin{equation}\widehat{f}^{(\mathrm{rec})}(\xi)=H\left(\frac1{2\epsilon}-|\xi|\right)\sum_{n\in\mathbb Z}\widehat f\left(\xi-\frac n\epsilon\right)\tag{9.128}\end{equation}
and, if $f$ is bandwidth limited with $\Xi<\frac1{2\epsilon}$, then $f(x)=f^{(\mathrm{rec})}(x)$ for all $x\in\mathbb R$.
\end{res}

\begin{res}{Proposition}{9.23}
For the different translation operators on each of the four domains $\mathbb Z/(N\mathbb Z)$, $\mathbb Z$, $\mathbb R/(L\mathbb Z)$ and $\mathbb R$, the Fourier modes on these domains are eigenvectors.
\end{res}

\section{Linear Differential Operators}

\begin{res}{Proposition}{10.1}
Let $\hat A$ be a densely defined operator on a Hilbert space $H$. For all $v\in\mathcal R_{\hat A}$ and all $w\in\ker(\hat A^\dagger)$, it holds that $\langle w,v\rangle=0$.
\end{res}

\begin{res}[Fredholm alternative]{Theorem}{10.2}
For a certain class of operators $\hat A$ on a Hilbert space $H$, a solution to the linear problem $\hat Au=v$ exists if and only if $v\perp\ker(\hat A^\dagger)$.
\end{res}

\begin{res}{Proposition}{10.3}
Functions $u,v\in L^2([a,b])$ with square integrable $p$th derivatives satisfy the \textbf{Lagrange identity}
\begin{equation}\overline{v(x)}(\hat Lu)(x)-\overline{(\hat L^\dagger v)(x)}u(x)=\frac{\mathrm d}{\mathrm dx}J(u(x),v(x))\tag{10.15}\end{equation}
where we have introduced the \textbf{bilinear concomitant} (it actually is sesquilinear)
\begin{equation}J(u(x),v(x))=\sum_{j=0}^p\sum_{k=0}^{j-1}(-1)^k\left(\frac{\mathrm d^k}{\mathrm dx^k}\bigl[a_j(x)\overline{v(x)}\bigr]\right)\left(\frac{\mathrm d^{j-1-k}}{\mathrm dx^{j-1-k}}u(x)\right).\tag{10.16}\end{equation}
\end{res}

\begin{res}{Corollary}{10.4}
These functions $u,v$ then also satisfy the \textbf{Green identity}
\begin{equation}\langle v,\hat Lu\rangle-\left\langle\hat L^\dagger v,u\right\rangle=J(u(x),v(x))\Big|_a^b=J(u(x),v(x))\Big|_{x=b}-J(u(x),v(x))\Big|_{x=a}.\tag{10.17}\end{equation}
\end{res}

\begin{res}{Proposition}{10.5}
A real second order differential operator is formally self-adjoint on $L^2_w([a,b])$ if and only if it takes the form of a \textbf{Sturm-Liouville operator}
\begin{equation}(\hat Lu)(x)=-\frac1{w(x)}\frac{\mathrm d}{\mathrm dx}\left(p(x)\frac{\mathrm d}{\mathrm dx}u(x)\right)+\frac{q(x)}{w(x)}u(x).\tag{10.26}\end{equation}
\end{res}

\begin{res}{Corollary}{10.6}
Consider a real second order differential operator $\hat L$ (as given in Eq.~(10.25) with $a_2(x)\ne0$ for all $x\in[a,b]$. There exists a weight function $w(x)>0$ so that $\hat L$ is a formally self-adjoint operator in $L^2_w([a,b])$.
\end{res}

\begin{res}{Proposition}{10.7}
A regular Sturm-Liouville operator is self-adjoint when imposing \textbf{separated boundary conditions}
\begin{equation}\mathsf B\boldsymbol x=\begin{bmatrix}\alpha_1&\alpha_2&0&0\\0&0&\beta_1&\beta_2\end{bmatrix}\begin{bmatrix}u(a)\\u'(a)\\u(b)\\u'(b)\end{bmatrix}=\boldsymbol 0\iff\begin{cases}\alpha_1u(a)+\alpha_2u'(a)=0\\\beta_1u(b)+\beta_2u'(b)=0\end{cases}.\tag{10.34}\end{equation}
\end{res}

\begin{res}{Proposition}{10.8}
A regular Sturm-Liouville operator with $p(a)=p(b)$ is self-adjoint when imposing \textbf{periodic boundary conditions}
\begin{equation}\mathsf B\boldsymbol x=\begin{bmatrix}1&0&-1&0\\0&1&0&-1\end{bmatrix}\begin{bmatrix}u(a)\\u'(a)\\u(b)\\u'(b)\end{bmatrix}=\boldsymbol 0\iff\begin{cases}u(a)=u(b)\\u'(a)=u'(b)\end{cases}\tag{10.35}\end{equation}
\end{res}

\begin{res}{Proposition}{10.9}
The solution to the homogeneous first-order vector valued initial value problem in Eq.~(10.38) is given by
\begin{equation}\boldsymbol z(t)=\sum_{n=0}^{+\infty}(\hat{\mathsf K}^n\boldsymbol\zeta)(t)=\sum_{n=0}^{+\infty}\int_c^t\mathrm dt_1\int_c^{t_1}\mathrm dt_2\cdots\int_c^{t_{n-1}}\mathrm dt_n\,\mathsf A(t_1)\mathsf A(t_2)\cdots\mathsf A(t_n)\boldsymbol\zeta\tag{10.46}\end{equation}
where, in order to apply the integral operator $\hat{\mathsf K}$ and powers thereof to the vector $\boldsymbol\zeta$, we need to interpret it as the constant vector-valued function $t\mapsto\boldsymbol\zeta$ for $t\in[a,b]$.
\end{res}

\begin{res}{Proposition}{10.10}
Let $\{\boldsymbol z_i:[a,b]\to\mathbb F^p;i=1,\dots,r\}$ be a set of solutions of the homogeneous first-order differential equation $\dot{\boldsymbol z}(t)=\mathsf A(t)\boldsymbol z(t)$ without boundary conditions, and denote the set of vectors $S(t)=\{\boldsymbol z_1(t),\dots,\boldsymbol z_r(t)\}$ obtained by evaluating the solutions at time $t$. The set of vectors $S(t_0)$ at a particular time $t_0$ is linearly independent if and only if $S(t)$ is linearly independent for any other time $t$.
\end{res}

\begin{res}{Proposition}{10.11}
The homogeneous first-order vector-valued differential equation $\dot{\boldsymbol z}(t)=\mathsf A(t)\boldsymbol z(t)$ without boundary conditions, where $\boldsymbol z(t)\in\mathbb F^p$, admits exactly $p$ linearly independent solutions, with respect to which any solution can be expanded.
\end{res}

\begin{res}[Liouville]{Theorem}{10.12}
Any matrix (function) satisfying $\dot{\mathsf Z}(t)=\mathsf A(t)\mathsf Z(t)$ satisfies
\begin{equation}\det\bigl(\mathsf Z(t)\bigr)=\det\bigl(\mathsf Z(c)\bigr)\exp\left[\int_c^t\operatorname{tr}\bigl(\mathsf A(\tau)\bigr)\,\mathrm d\tau\right].\tag{10.57}\end{equation}
\end{res}

\begin{res}{Proposition}{10.13}
Any two fundamental solution matrices $\mathsf Z(t)$ and $\tilde{\mathsf Z}(t)$ of the differential equation $\dot{\boldsymbol z}(t)=\mathsf A(t)\boldsymbol z(t)$ are related by a constant matrix via
\begin{equation}\tilde{\mathsf Z}(t)=\mathsf Z(t)\mathsf C\tag{10.60}\end{equation}
\end{res}

\begin{res}{Proposition}{10.14}
The principal fundamental matrix $\mathsf Z$ satisfies the following properties:
\begin{enumerate}
\item The solution of $\dot{\boldsymbol z}(t)=\mathsf A(t)\boldsymbol z(t)$ satisfying the initial condition $\boldsymbol z(t_0)=\boldsymbol\zeta$ is given by
\begin{equation}\boldsymbol z(t)=\mathsf Z(t,t_0)\boldsymbol\zeta.\tag{10.62}\end{equation}
\item For any other fundamental matrix $\tilde{\mathsf Z}$, it holds that $\mathsf Z(t,t_0)=\tilde{\mathsf Z}(t)\tilde{\mathsf Z}(t_0)^{-1}$.
\item The principal fundamental matrix satisfies the \textbf{stationarity property}
\begin{equation}\mathsf Z(t_2,t_1)\mathsf Z(t_1,t_0)=\mathsf Z(t_2,t_0)\tag{10.63}\end{equation}
and thus in particular $\mathsf Z(t_0,t_1)=\mathsf Z(t_1,t_0)^{-1}$.
\item $\det(\mathsf Z(t,t_0))=\exp\left[\int_{t_0}^t\operatorname{tr}(\mathsf A(\tau))\,\mathrm d\tau\right]$
\end{enumerate}
\end{res}

\begin{res}{Proposition}{10.15}
If the set of functions $\{u_i;i=1,\dots,p\}$ corresponds to a set of solutions of $\sum_{j=0}^pa_j(t)u^{(j)}(t)=0$ (with $a_p(t)\ne0$), then the associated Wronskian satisfies \textbf{Abel's formula}
\begin{equation}W(t)=W(t_0)\mathrm e^{-\int_{t_0}^t\frac{a_{p-1}(\tau)}{a_p(\tau)}\mathrm d\tau}.\tag{10.67}\end{equation}
\end{res}

\begin{res}{Proposition}{10.16}
Let $u$ be a solution of $a_2(t)\ddot u(t)+a_1(t)\dot u(t)+a_0(t)u(t)=0$. A linearly independent solution $v(t)$ is given by
\begin{equation}v(t)=u(t)\int_{t_0}^t\frac1{p(\tau)u(\tau)^2}\,\mathrm d\tau.\tag{10.68}\end{equation}
where $p(t)$ is defined (up to a constant factor) by $\frac{\mathrm d}{\mathrm dt}\log p(t)=\frac1{p(t)}\frac{\mathrm d}{\mathrm dt}p(t)=\frac{a_1(t)}{a_2(t)}$.
\end{res}

\begin{res}[Floquet]{Theorem}{10.17}
For a homogenous first order differential equation with $\mathsf A(t+T)=\mathsf A(t)$, every fundamental matrix can be expressed as
\begin{equation}\mathsf Z(t)=\mathsf Q(t)\mathrm e^{\mathsf Bt}\tag{10.70}\end{equation}
where $\mathsf Q(t)=\mathsf Q(t+T)$ is a periodic and $\mathsf B$ is constant.
\end{res}

\begin{res}{Proposition}{10.18}
The inhomogeneous initial value problem given in Eq.~(10.74) admits a unique solution given by
\begin{equation}\boldsymbol z(t)=\mathsf Z(t,t_0)\boldsymbol\zeta+\int_{t_0}^t\mathsf Z(t,\tau)\boldsymbol b(\tau)\,\mathrm d\tau.\tag{10.76}\end{equation}
\end{res}

\begin{res}{Theorem}{10.19}
If $\mathsf A(t)$ and $\boldsymbol b(t)$ are analytic (admit a Taylor expansion with a nonzero convergence radius) around some point $t_0$, then the solution $\boldsymbol z(t)$ of the inhomogeneous first order problem in Eq.~(10.74) is also analytic around $t_0$.
\end{res}

\begin{res}{Corollary}{10.20}
Any fundamental matrix $\mathsf Z(t)$ of the first order differential equation $\dot{\boldsymbol z}(t)=\mathsf A(t)\boldsymbol z(t)$ with $\mathsf A(t)$ analytic in $[a,b]$, will be analytic for all $t\in[a,b]$.
\end{res}

\begin{res}{Corollary}{10.21}
The solution $u(t)$ of the initial value problem in Eq.~(10.36) is analytic around $t=c$ if each of the coefficients $a_j(t)$ for $j=0,\dots,p$ and the right-hand side $f(t)$ are analytic around $t=c$, and furthermore $a_p(c)\ne0$.
\end{res}

\begin{res}{Proposition}{10.22}
Consider the first order homogeneous differential equation $\dot{\boldsymbol z}(t)=\mathsf A(t)\boldsymbol z(t)$ where $\mathsf A(t)$ is analytic in $(0,\rho)$ but has a singularity at $t=0$. The fundamental solution matrix is an analytic function of $(\log t)$ for $t\in(0,\rho)$.
\end{res}

\begin{res}{Theorem}{10.23}
Consider a first order homogeneous differential equation $\dot{\boldsymbol z}(t)=\mathsf A(t)\boldsymbol z(t)$ for which $t=0$ is a regular singular point. Any fundamental matrix solution takes the form $\mathsf Z(t)=\mathsf X(t)\mathrm e^{\mathsf B\log t}=\mathsf X(t)t^{\mathsf B}$ where $\mathsf X(t)$ admits a converging Taylor series expansion at $t=0$. If the spectrum of $\mathsf A_{-1}$ is such that no two distinct eigenvalues $\lambda_i\ne\lambda_j$ satisfy $\lambda_i-\lambda_j\in\mathbb Z$, then a fundamental matrix exists where $\mathsf X(0)=\mathsf I$ and $\mathsf B=\mathsf A_{-1}$.
\end{res}

\begin{res}{Theorem}{10.24}
Consider the boundary value problem $(\hat Lu)(x)=f(x)$ on the interval $[a,b]$, with $\hat L$ the $p$th order linear differential operator in Eq.~(10.7) and with boundary conditions $B_i[u]=\gamma_i$ for $i=1,\dots,p$ which are such that the solution $u_\gamma$ exits and is unique for all $\gamma_i$. For all $\xi\in[a,b]$, there exists a function $g_\xi(x)$, known as the \textbf{Green's function}, so that the solution $u_f$ satisfying the inhomogeneous differential equation with homogeneous boundary conditions can be written as
\begin{equation}u_f(x)=\int_a^bg_\xi(x)f(\xi)\,\mathrm d\xi\tag{10.113}\end{equation}
where $g_\xi(x)$ is completely specified by the conditions that for all $\xi\in(a,b)$
\begin{itemize}
\item $B_i[g_\xi]=0$ for $i=1,\dots,p$;
\item $(\hat Lg_\xi)(x)=0$ for $x\in(a,\xi)$ and $x\in(\xi,b)$;
\item $g_\xi(x)$ and its derivatives $g_\xi^{(j)}(x)$ for $j\le p-2$ are continuous at $x=\xi$;
\item $g_\xi^{(p-1)}(x)$ is discontinuous at $x=\xi$ and satisfies the `jump condition'
\begin{equation}g_\xi^{(p-1)}(\xi^+)-g_\xi^{(p-1)}(\xi^-)=\frac1{a_p(\xi)}.\tag{10.114}\end{equation}
\end{itemize}
\end{res}

\begin{res}{Theorem}{10.25}
For the regular Sturm-Liouville eigenvalue problem in Definition 10.4.1, the following properties hold:
\begin{enumerate}
\item The (point) spectrum of $\hat L$ corresponds to an infinite sequence of real numbers $\lambda_0<\lambda_1<\lambda_2<\ldots<\lambda_n<\ldots$, with some finite lower bound $-\infty<M<\lambda_0$ and $\lim_{n\to\infty}\lambda_n=+\infty$
\item For every eigenvalue $\lambda_n$, there is exactly one (linearly independent) eigenvector $u_n(x)$.
\item Eigenvectors corresponding to different eigenvalues are orthogonal with respect to the weighted inner product of $L^2_w([a,b])$.
\item The eigenvectors $\{u_n,n\in\mathbb N\}$ form a complete orthonormal basis.
\end{enumerate}
\end{res}

\begin{res}{Theorem}{10.26}
For functions $u\in\mathcal D_{\hat L}$, i.e.\ $u$ is continuous, differentiable and satisfies the boundary conditions, then the convergence of $u(x)=\sum_{n=0}^{+\infty}\langle u_n,u\rangle_wu_n(x)$ is uniform.
\end{res}

\begin{res}[Oscillation]{Theorem}{10.27}
The eigenfunction $u_n$ of the regular Sturm-Liouville eigenvalue problem in Definition 10.4.1 has exactly $n$ roots in $(a,b)$.
\end{res}

\begin{res}[Variational principle]{Theorem}{10.28}
Given a regular Sturm-Liouville operator, consider a general function $u\in\mathcal D_{\hat L}$ (thus in particular satisfying the boundary conditions). It holds that the \textbf{Rayleigh quotient}
\begin{equation}\mathcal R[u]=\frac{\langle u,\hat Lu\rangle_w}{\langle u,u\rangle_w}\tag{10.134}\end{equation}
satisfies $\mathcal R[u]\ge\lambda_0$, with the inequality becoming an equality if $u\sim u_0$.
\end{res}
